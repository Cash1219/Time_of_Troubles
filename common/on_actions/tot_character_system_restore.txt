on_game_started = {
	effect = {
			every_country = {
			tot_cleanup_idol_political_setup_effect = yes
			if = {
				limit = { NOT = { has_variable = tot_band_limit } }
				set_variable = { name = tot_band_limit value = 1 }
			}
			if = {
				limit = { var:tot_band_limit < 1 }
				set_variable = { name = tot_band_limit value = 1 }
			}
			tot_setup_default_kon_band_effect = yes
			tot_setup_default_story_bands_effect = yes
			tot_ensure_default_band_hq_slots_effect = yes
			tot_ensure_default_story_band_members_effect = yes
			update_kizuna = yes
			if = {
				limit = { is_player = yes }
				tot_music_competition_ensure_registration_cycle_effect = yes
			}
			if = { limit = { NOT = { has_variable = tot_music_competition_track } } set_variable = { name = tot_music_competition_track value = 1 } }
			if = {
				limit = { has_journal_entry = je_tot_club_gui }
				remove_journal_entry = je_tot_club_gui
			}
			if = {
				limit = {
					NOT = { has_journal_entry = je_tot_game_workshop_program }
					NOT = { has_variable = tot_game_workshop_program_complete }
					country_has_building_type_levels = {
						building = building_game
						value > 0
					}
				}
				add_journal_entry = { type = je_tot_game_workshop_program }
			}
			if = {
				limit = {
					NOT = { has_journal_entry = je_tot_goods_workshop_program }
					NOT = { has_variable = tot_goods_workshop_program_complete }
					country_has_building_type_levels = {
						building = building_goods
						value > 0
					}
				}
				add_journal_entry = { type = je_tot_goods_workshop_program }
			}
		}
	}
}

	on_monthly_pulse_country = {
	effect = {
		if = { limit = { this = c:RITA } tot_rita_mutsumi_leadership_monthly_process = yes }
		# Debt repayment must run in the country root scope, once per month.
		if = {
			limit = { c:AVMJ ?= this has_journal_entry = je_tot_avmj_debt_crisis }
			tot_avmj_debt_monthly_process = yes
		}
		tot_cleanup_idol_political_setup_effect = yes
		if = {
			limit = { has_journal_entry = je_tot_club_gui }
			remove_journal_entry = je_tot_club_gui
		}
		if = {
			limit = {
				NOT = { has_journal_entry = je_tot_game_workshop_program }
				NOT = { has_variable = tot_game_workshop_program_complete }
				country_has_building_type_levels = {
					building = building_game
					value > 0
				}
			}
			add_journal_entry = { type = je_tot_game_workshop_program }
		}
		if = {
			limit = {
				NOT = { has_journal_entry = je_tot_goods_workshop_program }
				NOT = { has_variable = tot_goods_workshop_program_complete }
				country_has_building_type_levels = {
					building = building_goods
					value > 0
				}
			}
			add_journal_entry = { type = je_tot_goods_workshop_program }
		}
		tot_band_initialize_axes = yes
		tot_ensure_default_band_hq_slots_effect = yes
		tot_ensure_default_story_band_members_effect = yes
		update_kizuna = yes
		tot_refresh_student_council_candidates_effect = yes
		tot_character_growth_refresh = yes
		tot_character_growth_monthly_update = yes
		tot_band_monthly_update = yes
		tot_gourmet_monthly_update = yes
		tot_music_competition_clear_invalid_band_entries_effect = yes
		# The player country owns the season state. This also repairs saves made
		# while the brief Vienna-host implementation was active.
		if = {
			limit = { is_player = yes }
			tot_music_competition_ensure_registration_cycle_effect = yes
			if = {
				limit = { has_variable = tot_music_competition_registration_open }
				tot_music_competition_build_field_effect = yes
				tot_music_competition_schedule_ai_registration_effect = yes
				tot_music_competition_start_if_ready_effect = yes
			}
			if = { limit = { has_variable = tot_music_competition_preparation_open } tot_music_competition_begin_after_preparation_effect = yes }
			tot_music_competition_schedule_weekly_tick_effect = yes
		}
		# Keep the public registration field current without requiring the player to
		# reopen the panel. Do not rebuild during scored stages, where rebuilding
		# would replace the active qualifying and final display lists.
		if = {
			limit = {
				is_player = yes
				has_variable = tot_music_competition_registration_open
			}
			tot_music_competition_build_field_effect = yes
			tot_music_competition_schedule_ai_registration_effect = yes
			if = { limit = { NOT = { has_variable = tot_music_competition_registration_months } } set_variable = { name = tot_music_competition_registration_months value = 0 } }
			change_variable = { name = tot_music_competition_registration_months add = 1 }
			tot_music_competition_start_if_ready_effect = yes
		}
		if = {
			limit = {
				is_player = yes
				has_variable = tot_music_competition_preparation_open
				NOT = { has_variable = tot_music_competition_stage }
			}
			tot_music_competition_build_field_effect = yes
			if = { limit = { NOT = { has_variable = tot_music_competition_preparation_months } } set_variable = { name = tot_music_competition_preparation_months value = 0 } }
			change_variable = { name = tot_music_competition_preparation_months add = 1 }
			tot_music_competition_begin_after_preparation_effect = yes
		}
		if = {
			limit = { is_player = yes has_variable = tot_music_competition_stage NOT = { has_variable = tot_music_competition_completed } }
			if = { limit = { var:tot_music_competition_stage = 1 NOT = { has_variable = tot_music_competition_qualifying_deadline } } set_variable = { name = tot_music_competition_qualifying_deadline weeks = 26 } }
			if = { limit = { var:tot_music_competition_stage = 2 NOT = { has_variable = tot_music_competition_final_deadline } } set_variable = { name = tot_music_competition_final_deadline weeks = 26 } }
			if = {
				limit = {
					var:tot_music_competition_stage = 2
					OR = {
						AND = { has_variable = tot_music_competition_final_days_remaining var:tot_music_competition_final_days_remaining <= 0 }
						AND = { has_variable = tot_music_competition_round var:tot_music_competition_round >= 26 }
					}
				}
				tot_music_competition_finish_final_effect = yes
			}
			else = { tot_music_competition_schedule_weekly_tick_effect = yes }
		}
	}
}
