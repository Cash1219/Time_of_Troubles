on_game_started = {
	effect = {
			every_country = {
			if = {
				limit = { NOT = { has_variable = tot_band_limit } }
				set_variable = { name = tot_band_limit value = 1 }
			}
			if = {
				limit = { var:tot_band_limit < 1 }
				set_variable = { name = tot_band_limit value = 1 }
			}
			tot_setup_default_kon_band_effect = yes
			tot_setup_default_story_bands_effect = yes
			tot_ensure_default_band_hq_slots_effect = yes
			tot_ensure_default_story_band_members_effect = yes
			update_kizuna = yes
			if = {
				limit = { has_journal_entry = je_tot_club_gui }
				remove_journal_entry = je_tot_club_gui
			}
			if = {
				limit = {
					NOT = { has_journal_entry = je_tot_game_workshop_program }
					NOT = { has_variable = tot_game_workshop_program_complete }
					country_has_building_type_levels = {
						building = building_game
						value > 0
					}
				}
				add_journal_entry = { type = je_tot_game_workshop_program }
			}
			if = {
				limit = {
					NOT = { has_journal_entry = je_tot_goods_workshop_program }
					NOT = { has_variable = tot_goods_workshop_program_complete }
					country_has_building_type_levels = {
						building = building_goods
						value > 0
					}
				}
				add_journal_entry = { type = je_tot_goods_workshop_program }
			}
		}
	}
}
