# A national-militia variant for the Gensokyo Community. The parent supplies
# the same ideological positions and mandatory-service technology gate.
law_tot_danmaku_warfare = {
	group = lawgroup_army_model
	parent = law_national_militia
	icon = "gfx/interface/icons/law_icons/national_militia.dds"
	progressiveness = 0
	unlocking_technologies = { mandatory_service }
	institution = institution_tot_lives
	institution_modifier = {
		country_tot_player_unit_capacity_add = 1
	}

	modifier = {
		country_institution_tot_lives_max_investment_add = 5
		unit_morale_loss_mult = -0.05
		state_conscription_rate_add = 0.05
		country_leverage_resistance_mult = 0.5
		country_officers_pol_str_mult = -0.25
		state_building_barrack_max_level_add = 5
		state_building_conscription_center_max_level_add = 100
		unit_kill_rate_add = -0.5
		country_diplomatic_play_maneuvers_mult = 0.5
		country_infamy_decay_mult = 0.1
	}

	can_impose = {
		OR = {
			AND = {
				is_in_same_power_bloc = scope:target_country
				modifier:country_can_impose_same_lawgroup_army_model_in_power_bloc_bool = yes
				has_law = scope:law
			}
			can_impose_law_default = yes
		}
	}

	is_visible = { tot_gensokyo_country_trigger = yes }
	can_enact = { tot_gensokyo_country_trigger = yes }

	disallowing_laws = {
		law_strict_edo_system
		law_intermediate_edo_system
	}

	on_enact = {
		if = {
			limit = { has_law_or_variant = law_type:law_peasant_levies }
			custom_tooltip = aristocrats_af_ig_shift_effect_negative_tt
		}
		if = {
			limit = { has_law_or_variant = law_type:law_professional_army }
			custom_tooltip = military_pops_af_ig_shift_effect_negative_tt
			custom_tooltip = non_military_pops_af_ig_shift_effect_negative_tt
		}
	}

	ai_will_do = {
		power_bloc ?= { has_principle = principle:principle_tot_gensokyo_autonomy_1 }
	}
}
