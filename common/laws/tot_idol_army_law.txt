# The three founding 2D-idol countries may adopt this professional-army variant.
# It inherits the parent law's ideological stances and keeps its military capacity.
law_idol_guard_army = {
	group = lawgroup_army_model
	parent = law_professional_army
	icon = "gfx/interface/icons/law_icons/professional_army.dds"

	progressiveness = 0
	modifier = {
		unit_morale_loss_mult = -0.1
		state_conscription_rate_add = 0.01
		state_building_barrack_max_level_add = 100
		state_building_conscription_center_max_level_add = 50
		country_tot_idol_guard_unit_ratio_add = 0.10
		country_tot_idol_knight_unit_ratio_add = 0.05
	}

	can_impose = {
		can_impose_law_default = yes
	}

	is_visible = {
		OR = {
			c:MUS ?= this
			c:IDMS ?= this
			c:GBDP ?= this
		}
	}

	disallowing_laws = {
		law_strict_edo_system
		law_intermediate_edo_system
	}

	on_enact = {
		if = {
			limit = { has_law_or_variant = law_type:law_peasant_levies }
			custom_tooltip = aristocrats_af_ig_shift_effect_negative_tt
		}
		if = {
			limit = { NOT = { has_law_or_variant = law_type:law_professional_army } }
			custom_tooltip = non_military_pops_af_ig_shift_effect_tt
			custom_tooltip = military_pops_af_ig_shift_effect_tt
		}
	}

	ai_will_do = { always = no }
}
