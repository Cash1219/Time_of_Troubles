tot_subject_development_common_1_AQO_button = {
	name = tot_subject_development_common_1_AQO_button
	desc = tot_subject_development_common_1_AQO_button_desc
	visible = {
		OR = {
			has_variable = tot_subject_development_AQO_member
			has_variable = tot_subject_development_AQO_overlord_member
		}
	}
	possible = {
		tot_subject_development_common_1_trigger = yes
		OR = {
			has_variable = tot_subject_development_AQO_member
			has_variable = tot_subject_development_AQO_overlord_member
		}
		NOT = { has_modifier = modifier_tot_subject_development_AQO_common_1_reward }
	}
	selected = {
		has_modifier = modifier_tot_subject_development_AQO_common_1_reward
	}
	effect = {
		if = {
			limit = { has_variable = tot_subject_development_AQO_member }
			scope:journal_entry = {
				change_variable = { name = tot_subject_development_AQO_common_completed_subject add = 1 }
				change_variable = { name = tot_subject_development_AQO_subject_score add = 1 }
			}
			add_modifier = {
				name = modifier_tot_subject_development_AQO_common_1_reward
				days = 3650
			}
		}
		else_if = {
			limit = { has_variable = tot_subject_development_AQO_overlord_member }
			scope:journal_entry = {
				change_variable = { name = tot_subject_development_AQO_common_completed_overlord add = 1 }
				change_variable = { name = tot_subject_development_AQO_overlord_score add = 1 }
			}
			add_modifier = {
				name = modifier_tot_subject_development_AQO_common_1_reward
				days = 3650
			}
		}
	}
	ai_chance = { value = 100 }
}
