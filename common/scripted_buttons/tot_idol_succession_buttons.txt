	name = je_tot_golden_hall_facade_button
	desc = je_tot_golden_hall_facade_button_desc

	visible = {
		OR = {
			has_journal_entry = je_tot_golden_hall_project
			has_journal_entry = je_tot_golden_hall_continuation
		}
	}

					c:GBDP ?= this
					c:HTT ?= this

		}
		tot_golden_hall_add_major_score = yes
		tot_golden_hall_check_core_completion = yes
	}
