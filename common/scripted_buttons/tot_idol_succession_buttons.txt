# The Golden Hall now advances through temporary capital construction contracts.
# Its old direct-investment buttons were intentionally removed.

je_tot_ll_stage_network_button = {
	name = je_tot_ll_stage_network_button
	desc = je_tot_ll_stage_network_button_desc

	visible = { has_journal_entry = je_tot_ll_publication_program }
	possible = {
		NOT = { has_variable = tot_ll_industry_stage_ready }
		any_scope_state = {
			state_region = s:STATE_CENTRAL_HUNGARY
			owner = root
			any_scope_building = { is_building_type = building_stage level >= 4 }
			any_scope_building = { is_building_type = building_university level >= 4 }
		}
	}
	effect = {
		set_variable = { name = tot_ll_industry_stage_ready value = 1 }
		every_country = {
			limit = { tot_golden_hall_crisis_participant_trigger = yes }
			change_variable = { name = tot_idol_crisis_ll_score add = 3 }
			change_variable = { name = tot_idol_crisis_compromise_score add = 1 }
		}
	}
	ai_chance = { base = 0 }
}
