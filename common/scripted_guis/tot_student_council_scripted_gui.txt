tot_appoint_student_council_president = {
	scope = character
	effect = {
		tot_appoint_student_council_president_effect = yes
	}
	is_shown = {
		is_character_alive = yes
		is_ruler_of_own_country = no
		is_heir_of_own_country = no
		is_interest_group_leader = no
		NOT = {
			has_role = character_role_tot_student_council_president
		}
		owner ?= {
			is_player = yes
			custom_tooltip = {
				text = TOT_CABINET_SEAT_REQUIRED
				OR = { tot_cabinet_seats_free >= 1 has_variable = tot_student_council_president }
			}
			custom_tooltip = {
				text = TOT_CHARACTER_CAPACITY_CORE_REQUIRED
				OR = { tot_club_capacity_free >= 30 has_variable = tot_student_council_president }
			}
		}
	}
	is_valid = {
		is_character_alive = yes
		custom_tooltip = {
			text = TOT_STUDENT_COUNCIL_NOT_RULER
			is_ruler_of_own_country = no
		}
		custom_tooltip = {
			text = TOT_STUDENT_COUNCIL_NOT_HEIR
			is_heir_of_own_country = no
		}
		custom_tooltip = {
			text = TOT_STUDENT_COUNCIL_NOT_INTEREST_GROUP_LEADER
			is_interest_group_leader = no
		}
		custom_tooltip = {
			text = TOT_STUDENT_COUNCIL_NOT_ALREADY_ASSIGNED
			NOT = {
				has_role = character_role_tot_student_council_president
			}
		}
		owner ?= {
			is_player = yes
			custom_tooltip = { text = TOT_CABINET_SEAT_REQUIRED OR = { tot_cabinet_seats_free >= 1 has_variable = tot_student_council_president } }
			custom_tooltip = { text = TOT_CHARACTER_CAPACITY_CORE_REQUIRED OR = { tot_club_capacity_free >= 30 has_variable = tot_student_council_president } }
		}
	}
}
