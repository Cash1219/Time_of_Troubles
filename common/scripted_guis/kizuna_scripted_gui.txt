get_is_in_government_type = {
	scope = character
	effect = {}
	is_shown = {
		this.interest_group ?= {
			is_in_government = yes
		}
	}
}
kon_kizuna_count = {
	scope = country
	effect = {
		
	}
	is_shown = {
		any_interest_group = {
				is_in_government = yes
				leader = {
					OR = {
						has_trait = tot_Kon_Yui_trait
						has_trait = tot_Kon_Mio_trait
						has_trait = tot_Kon_Tsumugi_trait
						has_trait = tot_Kon_Ritsu_trait
						has_trait = tot_Kon_Azusa_trait
					}
				}
			}
    } 		
    
	is_valid = {
		
	}
}
