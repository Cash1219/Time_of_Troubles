tot_band_hq_state_create = {
	scope = state
	effect = {
		owner ?= {
			if = {
				limit = { var:tot_band_limit >= 1 NOT = { has_variable = tot_band_1_created } }
				set_variable = { name = tot_band_1_created value = 1 }
				if = { limit = { NOT = { has_variable = tot_band_1_name } } set_variable = { name = tot_band_1_name value = 5 } }
				root = { set_variable = { name = tot_band_hq_slot value = 1 } }
			}
			else_if = {
				limit = { var:tot_band_limit >= 2 NOT = { has_variable = tot_band_2_created } }
				set_variable = { name = tot_band_2_created value = 1 }
				if = { limit = { NOT = { has_variable = tot_band_2_name } } set_variable = { name = tot_band_2_name value = 5 } }
				root = { set_variable = { name = tot_band_hq_slot value = 2 } }
			}
			else_if = {
				limit = { var:tot_band_limit >= 3 NOT = { has_variable = tot_band_3_created } }
				set_variable = { name = tot_band_3_created value = 1 }
				if = { limit = { NOT = { has_variable = tot_band_3_name } } set_variable = { name = tot_band_3_name value = 5 } }
				root = { set_variable = { name = tot_band_hq_slot value = 3 } }
			}
			update_kizuna = yes
		}
	}
	is_shown = {
		owner ?= { is_player = yes }
		any_scope_building = { is_building_type = building_band_headquarter }
		NOT = { has_variable = tot_band_hq_slot }
	}
	is_valid = {
		owner ?= {
			is_player = yes
			OR = {
				AND = { var:tot_band_limit >= 1 NOT = { has_variable = tot_band_1_created } }
				AND = { var:tot_band_limit >= 2 NOT = { has_variable = tot_band_2_created } }
				AND = { var:tot_band_limit >= 3 NOT = { has_variable = tot_band_3_created } }
			}
		}
		any_scope_building = { is_building_type = building_band_headquarter }
		NOT = { has_variable = tot_band_hq_slot }
	}
}
