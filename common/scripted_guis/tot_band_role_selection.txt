			update_kizuna = yes
		}
	}
	is_shown = {
		is_character_alive = yes
		every_scope_country = { limit = { is_player = yes } var:tot_band_limit >= 1 }
	}
	is_valid = {
		is_character_alive = yes
		every_scope_country = {
			limit = { is_player = yes }
			is_player = yes
			var:tot_band_limit >= 1
			custom_tooltip = {
				text = TOT_BAND_SELECT_ROLE_FIRST
				any_scope_state = { has_variable = tot_band_hq_slot var:tot_band_hq_slot = 1 has_variable = tot_band_selected_role }
			}
			custom_tooltip = {
				text = TOT_BAND_REWARD_POINT_COST
				OR = {
					has_variable = tot_band_reward_points_total
					var:tot_band_reward_points_total >= 1
				}
			}
			OR = {
				NOT = { has_variable = tot_band_1_member_count }
				NOT = { has_variable = tot_band_1_member_limit }
				var:tot_band_1_member_count < var:tot_band_1_member_limit

					text = TOT_BAND_RULE_ONE_VOCALIST

						NOT = { has_variable = tot_band_1_vocalist_count }
						var:tot_band_1_vocalist_count = 0

					}
				}
			}
			if = {
				limit = { any_scope_state = { has_variable = tot_band_hq_slot var:tot_band_hq_slot = 1 var:tot_band_selected_role = 2 } }
				custom_tooltip = {
					text = TOT_BAND_RULE_ONE_INSTRUMENT
					OR = {
						prev = {
							NOT = { has_variable = tot_band_role_guitarist }
							NOT = { has_variable = tot_band_role_bassist }
							NOT = { has_variable = tot_band_role_keyboardist }
							NOT = { has_variable = tot_band_role_drummer }
						}

				}
			}
			if = {

					}
				}
				custom_tooltip = {
					text = TOT_BAND_RULE_ONE_INSTRUMENT
					OR = {
						prev = {
							NOT = { has_variable = tot_band_role_guitarist }
							NOT = { has_variable = tot_band_role_bassist }
							NOT = { has_variable = tot_band_role_keyboardist }
							NOT = { has_variable = tot_band_role_drummer }
						}
						prev = { has_variable = tot_band_role_keyboardist has_variable = tot_band_id var:tot_band_id = 1 }
					}
				}
			}
			if = {
				limit = { any_scope_state = { has_variable = tot_band_hq_slot var:tot_band_hq_slot = 1 var:tot_band_selected_role = 5 } }
				custom_tooltip = {
