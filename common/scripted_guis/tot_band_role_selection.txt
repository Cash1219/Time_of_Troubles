	scope = character

		save_scope_as = tot_band_member
		owner ?= {
			if = { limit = { any_scope_state = { has_variable = tot_band_hq_slot var:tot_band_hq_slot = 1 has_variable = tot_band_selected_role var:tot_band_selected_role = 1 } } tot_appoint_band_1_vocalist_effect = yes }
			if = { limit = { any_scope_state = { has_variable = tot_band_hq_slot var:tot_band_hq_slot = 1 has_variable = tot_band_selected_role var:tot_band_selected_role = 2 } } tot_appoint_band_1_guitarist_effect = yes }
			if = { limit = { any_scope_state = { has_variable = tot_band_hq_slot var:tot_band_hq_slot = 1 has_variable = tot_band_selected_role var:tot_band_selected_role = 3 } } tot_appoint_band_1_bassist_effect = yes }
			if = { limit = { any_scope_state = { has_variable = tot_band_hq_slot var:tot_band_hq_slot = 1 has_variable = tot_band_selected_role var:tot_band_selected_role = 4 } } tot_appoint_band_1_keyboardist_effect = yes }
			if = { limit = { any_scope_state = { has_variable = tot_band_hq_slot var:tot_band_hq_slot = 1 has_variable = tot_band_selected_role var:tot_band_selected_role = 5 } } tot_appoint_band_1_drummer_effect = yes }
			update_kizuna = yes
		}
	}
	is_shown = {
		is_character_alive = yes
		owner ?= { is_player = yes var:tot_band_limit >= 1 }
	}
	is_valid = {
		is_character_alive = yes
		owner ?= {
			is_player = yes
			var:tot_band_limit >= 1
			custom_tooltip = {
				text = TOT_BAND_SELECT_ROLE_FIRST
				any_scope_state = { has_variable = tot_band_hq_slot var:tot_band_hq_slot = 1 has_variable = tot_band_selected_role }
			}
			OR = {
				NOT = { has_variable = tot_band_1_member_count }
				var:tot_band_1_member_count < 5
				prev = { has_variable = tot_band_id var:tot_band_id = 1 }
			}
			if = {
				limit = { any_scope_state = { has_variable = tot_band_hq_slot var:tot_band_hq_slot = 1 var:tot_band_selected_role = 1 } }
				custom_tooltip = {
					text = TOT_BAND_RULE_ONE_VOCALIST

						NOT = { has_variable = tot_band_1_vocalist_count }
						var:tot_band_1_vocalist_count = 0

					}
				}
			}
			if = {
				limit = { any_scope_state = { has_variable = tot_band_hq_slot var:tot_band_hq_slot = 1 var:tot_band_selected_role = 2 } }
				custom_tooltip = {
					text = TOT_BAND_RULE_ONE_INSTRUMENT
					OR = {
						prev = {
							NOT = { has_variable = tot_band_role_guitarist }
							NOT = { has_variable = tot_band_role_bassist }
							NOT = { has_variable = tot_band_role_keyboardist }
							NOT = { has_variable = tot_band_role_drummer }
						}
						prev = { has_variable = tot_band_role_guitarist has_variable = tot_band_id var:tot_band_id = 1 }
					}
				}
			}
			if = {
				limit = { any_scope_state = { has_variable = tot_band_hq_slot var:tot_band_hq_slot = 1 var:tot_band_selected_role = 3 } }
