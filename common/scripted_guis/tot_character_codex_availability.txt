# Codex directory visibility: current/subject use owned historical characters; faction/other use the global category.

tot_character_codex_show_001 = {
	scope = country
	effect = { }
	is_shown = {
		OR = {
			AND = {
				OR = { NOT = { has_variable = tot_character_codex_scope_mode } var:tot_character_codex_scope_mode = 1 }
				any_scope_character = { is_character_alive = yes is_historical = yes is_in_void = no is_in_exile_pool = no has_trait = tot_group_sheet_001_trait }
			}
			AND = {
				var:tot_character_codex_scope_mode = 2
				any_country = { is_subject_of = root any_scope_character = { is_character_alive = yes is_historical = yes is_in_void = no is_in_exile_pool = no has_trait = tot_group_sheet_001_trait } }
			}
			AND = { var:tot_character_codex_scope_mode = 3 }
		}
	}
	is_valid = { always = yes }
}
