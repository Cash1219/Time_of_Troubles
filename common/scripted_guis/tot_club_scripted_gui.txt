tot_club_refresh_panel = {
	scope = country
	effect = {
		tot_character_growth_refresh_club_bonus = yes
	}
	is_shown = { is_player = yes }
	is_valid = { is_player = yes }
}

tot_club_toggle_literature = {
	scope = country
	effect = {
		if = {
			limit = { has_variable = tot_club_theme_active_literature }
			remove_variable = tot_club_theme_active_literature
		}
		else = {
			set_variable = { name = tot_club_theme_active_literature value = 1 }
		}
		tot_character_growth_refresh_club_bonus = yes
	}
	is_shown = { is_player = yes }
	is_valid = {
		is_player = yes
		var:tot_club_theme_literature > 0
	}
}
