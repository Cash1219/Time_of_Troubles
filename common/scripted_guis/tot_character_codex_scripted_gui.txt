# This is deliberately a country-scoped GUI action. The card passes the
# selected character as a named scope, while root remains the player country.
tot_character_codex_recruit_subject = {
	scope = country
	effect = {
		scope:tot_subject_character = {
			owner ?= { add_liberty_desire = 50 }
			transfer_character = root
		}
		tot_character_codex_scan_selected_effect = yes
	}
	is_shown = {
		is_player = yes
		scope:tot_subject_character = {
			is_character_alive = yes
			is_in_void = no
			is_in_exile_pool = no
			is_ruler_of_own_country = no
			is_heir_of_own_country = no
			owner ?= { is_subject_of = root }
		}
	}
	is_valid = {
		is_player = yes
		scope:tot_subject_character = {
			is_character_alive = yes
			is_in_void = no
			is_in_exile_pool = no
			is_ruler_of_own_country = no
			is_heir_of_own_country = no
			owner ?= {
				is_subject_of = root
				custom_tooltip = {
					text = TOT_CHARACTER_CODEX_RECRUIT_SUBJECT_LIBERTY_REQUIREMENT
					liberty_desire <= 50
				}
			}
		}
	}
}

# The codex keeps its last selected bond while these controls change the
# left-hand directory category. Only the subject category narrows the scan.
tot_character_codex_scope_home = {
	scope = country
	effect = {
		set_variable = { name = tot_character_codex_scope_mode value = 1 }
		if = {
			limit = { has_variable = tot_character_codex_selection }
			tot_character_codex_scan_selected_effect = yes
		}
	}
	is_shown = { is_player = yes }
	is_valid = { is_player = yes }
}
