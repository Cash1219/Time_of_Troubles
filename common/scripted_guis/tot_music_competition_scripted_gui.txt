tot_music_competition_select_band = {
	scope = country
	effect = {
		set_variable = { name = tot_music_competition_track value = 1 }
		if = { limit = { NOT = { has_variable = tot_music_competition_band_registration_count } } set_variable = { name = tot_music_competition_band_registration_count value = 0 } }
		if = {
			limit = { NOT = { has_variable = tot_music_competition_band_id } }
			if = { limit = { has_variable = tot_band_1_created } set_variable = { name = tot_music_competition_band_id value = 1 } set_variable = { name = tot_music_competition_cancel_band_id value = 1 } }
			else_if = { limit = { has_variable = tot_band_2_created } set_variable = { name = tot_music_competition_band_id value = 2 } set_variable = { name = tot_music_competition_cancel_band_id value = 2 } }
			else_if = { limit = { has_variable = tot_band_3_created } set_variable = { name = tot_music_competition_band_id value = 3 } set_variable = { name = tot_music_competition_cancel_band_id value = 3 } }
			else_if = { limit = { has_variable = tot_band_4_created } set_variable = { name = tot_music_competition_band_id value = 4 } set_variable = { name = tot_music_competition_cancel_band_id value = 4 } }
			else_if = { limit = { has_variable = tot_band_5_created } set_variable = { name = tot_music_competition_band_id value = 5 } set_variable = { name = tot_music_competition_cancel_band_id value = 5 } }
		}
	}
	is_shown = { always = no }
	is_valid = { always = no }
}

tot_music_competition_initialize = {
	scope = country
			effect = {
			# Existing saves that had already reached the old closed-registration state
			# enter the new six-month preparation state rather than reopening entries.
			if = {
				limit = {
					has_variable = tot_music_competition_registered
					NOT = { has_variable = tot_music_competition_stage }
					has_variable = tot_music_competition_registration_closed
					NOT = { has_variable = tot_music_competition_preparation_open }
					NOT = { has_variable = tot_music_competition_spectator_view }
				}
				set_variable = { name = tot_music_competition_preparation_open value = 1 }
				set_variable = { name = tot_music_competition_preparation_months value = 0 }
				set_variable = { name = tot_music_competition_preparation_deadline months = 6 }
				set_variable = { name = tot_music_competition_preparation_days_remaining value = 182 }
			}
			# Registration and the actual competition are separate states. Older
			# saves without a stage are migrated into the new registration window.
			if = {
				limit = {
					has_variable = tot_music_competition_registered
					NOT = { has_variable = tot_music_competition_stage }
					NOT = { has_variable = tot_music_competition_registration_closed }
					NOT = { has_variable = tot_music_competition_registration_open }
					NOT = { has_variable = tot_music_competition_public_registration_open }
					NOT = { has_variable = tot_music_competition_spectator_view }
				}
			set_variable = { name = tot_music_competition_registration_open value = 1 }
			set_variable = { name = tot_music_competition_registration_months value = 0 }
			set_variable = { name = tot_music_competition_registration_deadline months = 12 }
			set_variable = { name = tot_music_competition_season_start_deadline months = 18 }
			set_variable = { name = tot_music_competition_registration_days_remaining value = 365 }
			remove_variable = tot_music_competition_registration_waiting_for_minimum
		}
		# Give pre-deadline saves a real remaining-time variable so the GUI can
		# display the calendar date rather than only a month counter.
		if = { limit = { has_variable = tot_music_competition_registered has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_deadline } has_variable = tot_music_competition_registration_months var:tot_music_competition_registration_months = 0 } set_variable = { name = tot_music_competition_registration_deadline months = 12 } }
		if = { limit = { has_variable = tot_music_competition_registered has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_deadline } has_variable = tot_music_competition_registration_months var:tot_music_competition_registration_months = 1 } set_variable = { name = tot_music_competition_registration_deadline months = 11 } }
		if = { limit = { has_variable = tot_music_competition_registered has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_deadline } has_variable = tot_music_competition_registration_months var:tot_music_competition_registration_months = 2 } set_variable = { name = tot_music_competition_registration_deadline months = 10 } }
		if = { limit = { has_variable = tot_music_competition_registered has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_deadline } has_variable = tot_music_competition_registration_months var:tot_music_competition_registration_months = 3 } set_variable = { name = tot_music_competition_registration_deadline months = 9 } }
		if = { limit = { has_variable = tot_music_competition_registered has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_deadline } has_variable = tot_music_competition_registration_months var:tot_music_competition_registration_months = 4 } set_variable = { name = tot_music_competition_registration_deadline months = 8 } }
		if = { limit = { has_variable = tot_music_competition_registered has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_deadline } has_variable = tot_music_competition_registration_months var:tot_music_competition_registration_months = 5 } set_variable = { name = tot_music_competition_registration_deadline months = 7 } }
		if = { limit = { has_variable = tot_music_competition_registered has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_deadline } has_variable = tot_music_competition_registration_months var:tot_music_competition_registration_months = 6 } set_variable = { name = tot_music_competition_registration_deadline months = 6 } }
		if = { limit = { has_variable = tot_music_competition_registered has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_deadline } has_variable = tot_music_competition_registration_months var:tot_music_competition_registration_months = 7 } set_variable = { name = tot_music_competition_registration_deadline months = 5 } }
		if = { limit = { has_variable = tot_music_competition_registered has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_deadline } has_variable = tot_music_competition_registration_months var:tot_music_competition_registration_months = 8 } set_variable = { name = tot_music_competition_registration_deadline months = 4 } }
		if = { limit = { has_variable = tot_music_competition_registered has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_deadline } has_variable = tot_music_competition_registration_months var:tot_music_competition_registration_months = 9 } set_variable = { name = tot_music_competition_registration_deadline months = 3 } }
		if = { limit = { has_variable = tot_music_competition_registered has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_deadline } has_variable = tot_music_competition_registration_months var:tot_music_competition_registration_months = 10 } set_variable = { name = tot_music_competition_registration_deadline months = 2 } }
		if = { limit = { has_variable = tot_music_competition_registered has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_deadline } has_variable = tot_music_competition_registration_months var:tot_music_competition_registration_months = 11 } set_variable = { name = tot_music_competition_registration_deadline months = 1 } }
		# Old active saves did not record a calendar deadline for the qualifying phase.
		if = { limit = { has_variable = tot_music_competition_stage var:tot_music_competition_stage = 1 NOT = { has_variable = tot_music_competition_qualifying_deadline } } set_variable = { name = tot_music_competition_qualifying_deadline weeks = 26 } }
		if = { limit = { has_variable = tot_music_competition_stage var:tot_music_competition_stage = 2 NOT = { has_variable = tot_music_competition_final_deadline } } set_variable = { name = tot_music_competition_final_deadline weeks = 26 } }
		# Convert pre-countdown saves once. The monthly counter provides a stable
		# approximation; all new seasons use the exact weekly counter from day one.
		if = { limit = { has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_days_remaining } has_variable = tot_music_competition_registration_months var:tot_music_competition_registration_months = 0 } set_variable = { name = tot_music_competition_registration_days_remaining value = 365 } }
		if = { limit = { has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_days_remaining } has_variable = tot_music_competition_registration_months var:tot_music_competition_registration_months = 1 } set_variable = { name = tot_music_competition_registration_days_remaining value = 335 } }
		if = { limit = { has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_days_remaining } has_variable = tot_music_competition_registration_months var:tot_music_competition_registration_months = 2 } set_variable = { name = tot_music_competition_registration_days_remaining value = 305 } }
		if = { limit = { has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_days_remaining } has_variable = tot_music_competition_registration_months var:tot_music_competition_registration_months = 3 } set_variable = { name = tot_music_competition_registration_days_remaining value = 275 } }
		if = { limit = { has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_days_remaining } has_variable = tot_music_competition_registration_months var:tot_music_competition_registration_months = 4 } set_variable = { name = tot_music_competition_registration_days_remaining value = 245 } }
		if = { limit = { has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_days_remaining } has_variable = tot_music_competition_registration_months var:tot_music_competition_registration_months = 5 } set_variable = { name = tot_music_competition_registration_days_remaining value = 215 } }
		if = { limit = { has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_days_remaining } has_variable = tot_music_competition_registration_months var:tot_music_competition_registration_months = 6 } set_variable = { name = tot_music_competition_registration_days_remaining value = 185 } }
		if = { limit = { has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_days_remaining } has_variable = tot_music_competition_registration_months var:tot_music_competition_registration_months = 7 } set_variable = { name = tot_music_competition_registration_days_remaining value = 155 } }
		if = { limit = { has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_days_remaining } has_variable = tot_music_competition_registration_months var:tot_music_competition_registration_months = 8 } set_variable = { name = tot_music_competition_registration_days_remaining value = 125 } }
		if = { limit = { has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_days_remaining } has_variable = tot_music_competition_registration_months var:tot_music_competition_registration_months = 9 } set_variable = { name = tot_music_competition_registration_days_remaining value = 95 } }
		if = { limit = { has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_days_remaining } has_variable = tot_music_competition_registration_months var:tot_music_competition_registration_months = 10 } set_variable = { name = tot_music_competition_registration_days_remaining value = 65 } }
		if = { limit = { has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_days_remaining } has_variable = tot_music_competition_registration_months var:tot_music_competition_registration_months = 11 } set_variable = { name = tot_music_competition_registration_days_remaining value = 35 } }
		if = { limit = { has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_days_remaining } has_variable = tot_music_competition_registration_months var:tot_music_competition_registration_months >= 12 } set_variable = { name = tot_music_competition_registration_days_remaining value = 0 } }
		if = { limit = { has_variable = tot_music_competition_registration_open NOT = { has_variable = tot_music_competition_registration_days_remaining } NOT = { has_variable = tot_music_competition_registration_months } } set_variable = { name = tot_music_competition_registration_days_remaining value = 365 } }
		if = { limit = { has_variable = tot_music_competition_preparation_open NOT = { has_variable = tot_music_competition_preparation_days_remaining } } set_variable = { name = tot_music_competition_preparation_days_remaining value = 182 } }
		if = { limit = { has_variable = tot_music_competition_stage var:tot_music_competition_stage = 1 NOT = { has_variable = tot_music_competition_qualifying_days_remaining } } set_variable = { name = tot_music_competition_qualifying_days_remaining value = 182 } }
		if = { limit = { has_variable = tot_music_competition_stage var:tot_music_competition_stage = 2 NOT = { has_variable = tot_music_competition_final_days_remaining } } set_variable = { name = tot_music_competition_final_days_remaining value = 182 } }
		# Apply elapsed phase boundaries immediately when this panel is opened.
		if = { limit = { has_variable = tot_music_competition_registration_open has_variable = tot_music_competition_registration_days_remaining var:tot_music_competition_registration_days_remaining <= 0 } tot_music_competition_start_if_ready_effect = yes }
		if = { limit = { has_variable = tot_music_competition_preparation_open has_variable = tot_music_competition_preparation_days_remaining var:tot_music_competition_preparation_days_remaining <= 0 } tot_music_competition_begin_after_preparation_effect = yes }
		# Recover an old season that reached an elapsed boundary with a stale stop flag.
		if = {
			limit = {
				has_variable = tot_music_competition_completed
				OR = {
					AND = { has_variable = tot_music_competition_registration_open has_variable = tot_music_competition_registration_days_remaining var:tot_music_competition_registration_days_remaining <= 0 }
					AND = { has_variable = tot_music_competition_preparation_open has_variable = tot_music_competition_preparation_days_remaining var:tot_music_competition_preparation_days_remaining <= 0 }
				}
			}
			remove_variable = tot_music_competition_completed
			remove_variable = tot_music_competition_weekly_tick_scheduled
			remove_variable = tot_music_competition_next_tick
			tot_music_competition_start_if_ready_effect = yes
			tot_music_competition_begin_after_preparation_effect = yes
		}
		# Older builds resolved the champion one week early. Remove that stale result
		# from any active save, even if it has already received a prior migration flag.
		if = {
			limit = {
				has_variable = tot_music_competition_track
				var:tot_music_competition_track = 1
				has_variable = tot_music_competition_stage
				var:tot_music_competition_stage = 2
				NOT = { has_variable = tot_music_competition_completed }
				has_variable = tot_music_competition_final_champion_resolved
				has_variable = tot_music_competition_round
				var:tot_music_competition_round < 26
			}
			remove_variable = tot_music_competition_final_champion_resolved
			tot_music_competition_clear_final_slot_effect = { SLOT = tot_music_competition_final_champion }
		}
		# The original final phase only ran a timer. Complete its bracket once when
		# opening an affected save, even though that timer already marked the season done.
		if = {
			limit = {
				NOT = { has_variable = tot_music_competition_final_timing_v2 }
				has_variable = tot_music_competition_track
				var:tot_music_competition_track = 1
				has_variable = tot_music_competition_stage
				var:tot_music_competition_stage = 2
				NOT = { has_variable = tot_music_competition_completed }
			}
			# Old builds resolved a champion before the final week. Keep the earlier bracket
			# results, but suppress that premature champion until the season ends.
			if = {
				limit = {
					has_variable = tot_music_competition_final_champion_resolved
					has_variable = tot_music_competition_round
					var:tot_music_competition_round < 26
				}
				remove_variable = tot_music_competition_final_champion_resolved
				tot_music_competition_clear_final_slot_effect = { SLOT = tot_music_competition_final_champion }
			}
			set_variable = { name = tot_music_competition_final_timing_v2 value = 1 }
		}
		# The first bracket version advanced only the left-side winners, dropping
		# the winners of matches two and four. Rebuild any in-progress band final
		# once from its existing seeded field with the pairwise score comparison.
		if = {
			limit = {
				NOT = { has_variable = tot_music_competition_final_match_logic_v2 }
				has_variable = tot_music_competition_track
				var:tot_music_competition_track = 1
				has_variable = tot_music_competition_stage
				var:tot_music_competition_stage = 2
				NOT = { has_variable = tot_music_competition_completed }
			}
			tot_music_competition_clear_final_bracket_effect = yes
			set_variable = { name = tot_music_competition_final_match_logic_v2 value = 1 }
			tot_music_competition_resolve_band_final_effect = yes
		}
		if = {
			limit = {
				has_variable = tot_music_competition_stage
				var:tot_music_competition_stage = 2
				has_variable = tot_music_competition_completed
				has_variable = tot_music_competition_track
				var:tot_music_competition_track = 1
				NOT = { has_variable = tot_music_competition_final_champion_resolved }
			}
			tot_music_competition_resolve_band_final_effect = yes
		}
		# Finish stalled legacy finals as soon as the player opens the activity page.
		if = {
			limit = {
				has_variable = tot_music_competition_stage
				var:tot_music_competition_stage = 2
				NOT = { has_variable = tot_music_competition_completed }
				has_variable = tot_music_competition_final_days_remaining
				OR = {
					var:tot_music_competition_final_days_remaining <= 0
					AND = { has_variable = tot_music_competition_round var:tot_music_competition_round >= 26 }
				}
			}
			tot_music_competition_finish_final_effect = yes
		}
		# Versioned recovery for saves made while the old scheduler could leave a
		# stale flag behind. It runs once, then normal weekly event chaining owns
		# the timer without re-scheduling on every panel open.
		if = {
			limit = {
				NOT = { has_variable = tot_music_competition_tick_recovery_v2 }
				OR = {
					has_variable = tot_music_competition_registration_open
					has_variable = tot_music_competition_preparation_open
					has_variable = tot_music_competition_stage
				}
			}
			remove_variable = tot_music_competition_weekly_tick_scheduled
			remove_variable = tot_music_competition_next_tick
			set_variable = { name = tot_music_competition_tick_recovery_v2 value = 1 }
		}
		if = {
				limit = { NOT = { has_variable = tot_music_competition_track } }
				set_variable = { name = tot_music_competition_track value = 1 }
			}
			if = {
				limit = { has_variable = tot_music_competition_track var:tot_music_competition_track = 1 NOT = { has_variable = tot_music_competition_band_id } }
				if = { limit = { has_variable = tot_band_1_created } set_variable = { name = tot_music_competition_band_id value = 1 } }
				else_if = { limit = { has_variable = tot_band_2_created } set_variable = { name = tot_music_competition_band_id value = 2 } }
				else_if = { limit = { has_variable = tot_band_3_created } set_variable = { name = tot_music_competition_band_id value = 3 } }
				else_if = { limit = { has_variable = tot_band_4_created } set_variable = { name = tot_music_competition_band_id value = 4 } }
				else_if = { limit = { has_variable = tot_band_5_created } set_variable = { name = tot_music_competition_band_id value = 5 } }
			}
			if = {
				limit = {
					has_variable = tot_music_competition_track
					var:tot_music_competition_track = 1
					has_variable = tot_music_competition_band_id
					OR = {
						AND = { var:tot_music_competition_band_id = 1 NOT = { has_variable = tot_band_1_created } }
						AND = { var:tot_music_competition_band_id = 2 NOT = { has_variable = tot_band_2_created } }
						AND = { var:tot_music_competition_band_id = 3 NOT = { has_variable = tot_band_3_created } }
						AND = { var:tot_music_competition_band_id = 4 NOT = { has_variable = tot_band_4_created } }
						AND = { var:tot_music_competition_band_id = 5 NOT = { has_variable = tot_band_5_created } }
					}
				}
				remove_variable = tot_music_competition_band_id
				if = { limit = { has_variable = tot_band_1_created } set_variable = { name = tot_music_competition_band_id value = 1 } }
				else_if = { limit = { has_variable = tot_band_2_created } set_variable = { name = tot_music_competition_band_id value = 2 } }
				else_if = { limit = { has_variable = tot_band_3_created } set_variable = { name = tot_music_competition_band_id value = 3 } }
				else_if = { limit = { has_variable = tot_band_4_created } set_variable = { name = tot_music_competition_band_id value = 4 } }
				else_if = { limit = { has_variable = tot_band_5_created } set_variable = { name = tot_music_competition_band_id value = 5 } }
			}
			# Older saves selected a band without persisting the withdrawal target.
			# Re-sync it once when the panel opens so cancellation always targets the
			# highlighted band rather than a stale last selection.
			if = { limit = { has_variable = tot_music_competition_track var:tot_music_competition_track = 1 has_variable = tot_music_competition_band_id var:tot_music_competition_band_id = 1 } set_variable = { name = tot_music_competition_cancel_band_id value = 1 } }
			if = { limit = { has_variable = tot_music_competition_track var:tot_music_competition_track = 1 has_variable = tot_music_competition_band_id var:tot_music_competition_band_id = 2 } set_variable = { name = tot_music_competition_cancel_band_id value = 2 } }
			if = { limit = { has_variable = tot_music_competition_track var:tot_music_competition_track = 1 has_variable = tot_music_competition_band_id var:tot_music_competition_band_id = 3 } set_variable = { name = tot_music_competition_cancel_band_id value = 3 } }
			if = { limit = { has_variable = tot_music_competition_track var:tot_music_competition_track = 1 has_variable = tot_music_competition_band_id var:tot_music_competition_band_id = 4 } set_variable = { name = tot_music_competition_cancel_band_id value = 4 } }
			if = { limit = { has_variable = tot_music_competition_track var:tot_music_competition_track = 1 has_variable = tot_music_competition_band_id var:tot_music_competition_band_id = 5 } set_variable = { name = tot_music_competition_cancel_band_id value = 5 } }
		# Upgrade already-registered saves made before first-round match lists existed.
		if = {
			limit = {
				has_variable = tot_music_competition_registered
				has_variable = tot_music_competition_round
				var:tot_music_competition_round = 1
			}
			if = { limit = { NOT = { has_variable = tot_music_competition_stage } } set_variable = { name = tot_music_competition_stage value = 1 } }
			tot_music_competition_build_field_effect = yes
		}
		# The field is rebuilt by registration/cancellation and phase transitions.
		# Do not rebuild it just because the panel was opened again: that made a
		# valid public roster look as if it had been reset.
		# Recover display slots from the registered teams if a previous GUI parse
		# failure left an active qualifying screen blank. Scores stay untouched.
		if = {
			limit = {
				has_variable = tot_music_competition_track
				var:tot_music_competition_track = 1
				has_variable = tot_music_competition_stage
				var:tot_music_competition_stage = 1
				NOT = { has_variable = tot_music_competition_completed }
				NOT = { has_variable_list = tot_music_competition_display_slot_01_band_1 }
			}
			tot_music_competition_build_band_team_field_effect = yes
		}
		# Rebuild the opening knockout seeds whenever an unplayed band final is opened.
		# This also repairs finals created by the earlier, unseeded bracket code.
		if = {
			limit = {
				has_variable = tot_music_competition_track
				var:tot_music_competition_track = 1
				has_variable = tot_music_competition_stage
				var:tot_music_competition_stage = 2
				has_variable = tot_music_competition_round
				var:tot_music_competition_round = 0
			}
			tot_music_competition_build_qualified_band_field_effect = yes
		}
