	scope = character
	effect = {
		owner ?= {
			set_variable = { name = tot_character_growth_selected_progress_character value = prev }
		}
	}
	is_shown = { is_character_alive = yes historical = yes owner ?= { is_player = yes } }
	is_valid = { is_character_alive = yes historical = yes owner ?= { is_player = yes } }
}

	scope = country

	is_shown = {
		has_variable = tot_character_growth_selected_progress_character
		var:tot_character_growth_selected_progress_character ?= { is_character_alive = yes }
	}
	is_valid = {
		has_variable = tot_character_growth_selected_progress_character
		var:tot_character_growth_selected_progress_character ?= { is_character_alive = yes }
	}
}

tot_character_growth_upgrade_guitarist = {
	scope = character
	effect = {
		change_variable = { name = tot_character_growth_skill_points add = -1 }
		if = { limit = { NOT = { has_variable = tot_character_growth_guitarist_mastery } } set_variable = { name = tot_character_growth_guitarist_mastery value = 1 } }
		else = { change_variable = { name = tot_character_growth_guitarist_mastery add = 1 } }
	}
	is_shown = { OR = { has_trait = tot_guitarist_trait has_trait = tot_is_guitarist_trait } }
	is_valid = {
		OR = { has_trait = tot_guitarist_trait has_trait = tot_is_guitarist_trait }
		has_variable = tot_character_growth_skill_points
		var:tot_character_growth_skill_points >= 1
		OR = { NOT = { has_variable = tot_character_growth_guitarist_mastery } var:tot_character_growth_guitarist_mastery < 5 }
	}
}

tot_character_growth_buy_random_trait = {
	scope = character
	effect = {
		change_variable = { name = tot_character_growth_skill_points add = -2 }
		tot_character_growth_grant_random_kizuna_trait = yes
		owner ?= { update_kizuna = yes }
	}
	is_shown = { is_character_alive = yes historical = yes owner ?= { is_player = yes } }
	is_valid = {
		is_character_alive = yes
		historical = yes
		owner ?= { is_player = yes }
		has_variable = tot_character_growth_skill_points
		var:tot_character_growth_skill_points >= 2
		tot_character_growth_has_available_random_trait = yes
	}
}
