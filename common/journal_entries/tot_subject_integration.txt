# 二偶企划发展竞赛: one JE per affiliate (added via its country history), overlord hardcoded as co-participant
je_tot_idol_subject_development_AQO = {
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	group = je_group_global_international_situations
	should_be_pinned_by_default_involved = yes
	should_be_pinned_by_default_uninvolved_or_context = no
	weight = 10000
	timeout = 3650

	should_be_involved = {
		OR = {
			c:AQO ?= this
			c:MUS ?= this
		}
	}
	should_show_when_not_involved = {
		always = no
	}


	scripted_button = tot_subject_development_common_1_AQO_button
	scripted_button = tot_subject_development_common_2_AQO_button
	scripted_button = tot_subject_development_common_3_AQO_button
	scripted_button = tot_subject_development_common_4_AQO_button
	scripted_button = tot_subject_development_common_5_AQO_button
	scripted_button = tot_subject_development_overlord_1_AQO_button
	scripted_button = tot_subject_development_overlord_2_AQO_button
	scripted_button = tot_subject_development_overlord_3_AQO_button
	scripted_button = tot_subject_development_national_1_AQO_button
	scripted_button = tot_subject_development_national_2_AQO_button
	scripted_button = tot_subject_development_national_3_AQO_button

	immediate = {
		c:AQO ?= {
			set_variable = tot_subject_development_AQO_member
			save_scope_as = tot_subject_development_subject
		}
		c:MUS ?= {
			set_variable = tot_subject_development_AQO_overlord_member
			save_scope_as = tot_subject_development_overlord
		}
	}

	on_monthly_pulse = {
		effect = {
			c:AQO ?= {
				if = {
					limit = { NOT = { has_variable = tot_subject_development_AQO_gdp_goal } }
					set_variable = {
						name = tot_subject_development_AQO_gdp_goal
						value = {
							value = gdp
							multiply = 1.5
						}
					}
				}
				if = {
					limit = { NOT = { has_variable = tot_subject_development_AQO_member } }
					set_variable = tot_subject_development_AQO_member
					save_scope_as = tot_subject_development_subject
				}
			}
			c:MUS ?= {
				if = {
					limit = { NOT = { has_variable = tot_subject_development_AQO_overlord_member } }
					set_variable = tot_subject_development_AQO_overlord_member
					save_scope_as = tot_subject_development_overlord
				}
			}
			scope:journal_entry = {
				if = {
					limit = { NOT = { has_variable = tot_subject_development_AQO_inited } }
					set_variable = tot_subject_development_AQO_inited
					set_variable = { name = tot_subject_development_AQO_common_completed_subject value = 0 }
					set_variable = { name = tot_subject_development_AQO_common_completed_overlord value = 0 }
					set_variable = { name = tot_subject_development_AQO_overlord_completed value = 0 }
					set_variable = { name = tot_subject_development_AQO_national_completed value = 0 }
					set_variable = { name = tot_subject_development_AQO_subject_score value = 0 }
					set_variable = { name = tot_subject_development_AQO_overlord_score value = 0 }
				}
				if = {
					limit = {
						NOT = {
							any_scope_je_involved = {
								has_variable = tot_subject_development_AQO_overlord_member
							}
						}
					}
					add_involved_country = c:MUS
				}
			}
		}
	}

	complete = {
		scope:journal_entry = {
			has_variable = tot_subject_development_AQO_inited
			var:tot_subject_development_AQO_subject_score >= 5
			var:tot_subject_development_AQO_subject_score > var:tot_subject_development_AQO_overlord_score
		}
	}

	fail = {
		scope:journal_entry = {
			has_variable = tot_subject_development_AQO_inited
			var:tot_subject_development_AQO_overlord_score >= 5
			var:tot_subject_development_AQO_overlord_score > var:tot_subject_development_AQO_subject_score
		}
	}

	on_complete = {
		scope:journal_entry = {
			set_variable = tot_subject_development_AQO_finished
			custom_tooltip = tot_subject_development_subject_winner_tt
			every_scope_je_involved = {
				if = {
					limit = { has_variable = tot_subject_development_AQO_member }
					change_subject_type = subject_type_dominion
					add_liberty_desire = 10
					add_modifier = {
						name = modifier_tot_subject_development_subject_winner
						days = long_modifier_time
					}
					add_modifier = {
						name = modifier_tot_subject_development_participant
						days = normal_modifier_time
					}
					set_variable = tot_subject_development_AQO_finished
					remove_variable = tot_subject_development_AQO_member
				}
				else = {
					if = {
						limit = { has_variable = tot_subject_development_AQO_overlord_member }
						add_modifier = {
							name = modifier_tot_subject_development_overlord_loser
							days = normal_modifier_time
						}
						add_modifier = {
							name = modifier_tot_subject_development_participant
							days = normal_modifier_time
						}
						set_variable = tot_subject_development_AQO_finished
						remove_variable = tot_subject_development_AQO_overlord_member
					}
				}
			}
		}
	}

	on_fail = {
		scope:journal_entry = {
			set_variable = tot_subject_development_AQO_finished
			custom_tooltip = tot_subject_development_overlord_winner_tt
			every_scope_je_involved = {
				if = {
					limit = { has_variable = tot_subject_development_AQO_member }
					change_subject_type = subject_type_puppet
					add_liberty_desire = -10
					add_modifier = {
						name = modifier_tot_subject_development_participant
						days = normal_modifier_time
					}
					set_variable = tot_subject_development_AQO_finished
					remove_variable = tot_subject_development_AQO_member
				}
				else = {
					if = {
						limit = { has_variable = tot_subject_development_AQO_overlord_member }
						add_modifier = {
							name = modifier_tot_subject_development_overlord_winner
							days = long_modifier_time
						}
						add_modifier = {
							name = modifier_tot_subject_development_participant
							days = normal_modifier_time
						}
						set_variable = tot_subject_development_AQO_finished
						remove_variable = tot_subject_development_AQO_overlord_member
					}
				}
			}
		}
	}

	on_timeout = {
		scope:journal_entry = {
			set_variable = tot_subject_development_AQO_finished
			custom_tooltip = tot_subject_development_timeout_tt
			if = {
				limit = { var:tot_subject_development_AQO_subject_score > var:tot_subject_development_AQO_overlord_score }
				every_scope_je_involved = {
					if = {
						limit = { has_variable = tot_subject_development_AQO_member }
						change_subject_type = subject_type_dominion
						add_liberty_desire = 10
						add_modifier = {
							name = modifier_tot_subject_development_subject_winner
							days = long_modifier_time
						}
						add_modifier = {
							name = modifier_tot_subject_development_participant
							days = normal_modifier_time
						}
						set_variable = tot_subject_development_AQO_finished
						remove_variable = tot_subject_development_AQO_member
					}
					else = {
						if = {
							limit = { has_variable = tot_subject_development_AQO_overlord_member }
							add_modifier = {
								name = modifier_tot_subject_development_overlord_loser
								days = normal_modifier_time
							}
							add_modifier = {
								name = modifier_tot_subject_development_participant
								days = normal_modifier_time
							}
							set_variable = tot_subject_development_AQO_finished
							remove_variable = tot_subject_development_AQO_overlord_member
						}
					}
				}
			}
			else_if = {
				limit = { var:tot_subject_development_AQO_overlord_score > var:tot_subject_development_AQO_subject_score }
				every_scope_je_involved = {
					if = {
						limit = { has_variable = tot_subject_development_AQO_member }
						change_subject_type = subject_type_puppet
						add_liberty_desire = -10
						add_modifier = {
							name = modifier_tot_subject_development_participant
							days = normal_modifier_time
						}
						set_variable = tot_subject_development_AQO_finished
						remove_variable = tot_subject_development_AQO_member
					}
					else = {
						if = {
							limit = { has_variable = tot_subject_development_AQO_overlord_member }
							add_modifier = {
								name = modifier_tot_subject_development_overlord_winner
								days = long_modifier_time
							}
							add_modifier = {
								name = modifier_tot_subject_development_participant
								days = normal_modifier_time
							}
							set_variable = tot_subject_development_AQO_finished
							remove_variable = tot_subject_development_AQO_overlord_member
						}
					}
				}
			}
			else = {
				every_scope_je_involved = {
					if = {
						limit = { has_variable = tot_subject_development_AQO_member }
						change_subject_type = subject_type_protectorate
						add_modifier = {
							name = modifier_tot_subject_development_timeout
							days = normal_modifier_time
						}
						add_modifier = {
							name = modifier_tot_subject_development_participant
							days = normal_modifier_time
						}
						set_variable = tot_subject_development_AQO_finished
						remove_variable = tot_subject_development_AQO_member
					}
					else = {
						if = {
							limit = { has_variable = tot_subject_development_AQO_overlord_member }
							add_modifier = {
								name = modifier_tot_subject_development_timeout
								days = normal_modifier_time
							}
							add_modifier = {
								name = modifier_tot_subject_development_participant
								days = normal_modifier_time
							}
							set_variable = tot_subject_development_AQO_finished
							remove_variable = tot_subject_development_AQO_overlord_member
						}
					}
				}
			}
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_tot_idol_subject_development_status
				trigger = { always = yes }
			}
		}
	}
}

				}

					limit = {
						NOT = {
							any_scope_je_involved = {
								has_variable = tot_subject_development_UKR_overlord_member
							}
						}
