je_tot_avmj_debt_crisis = {
	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	group = je_group_tot_idol_succession
	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	weight = 10000

	is_shown_in_lobby = { c:AVMJ ?= this }
	should_be_involved = {
		OR = {
			c:AVMJ ?= this
			AND = { c:AVMJ ?= { var:tot_avmj_htt_claim > 0 } c:HTT ?= this }
			AND = { c:AVMJ ?= { var:tot_avmj_gbdp_claim > 0 } c:GBDP ?= this }
			AND = { c:AVMJ ?= { var:tot_avmj_mygo_claim > 0 var:tot_avmj_mygo_claim_holder = 0 } c:MYG ?= this }
			AND = { c:AVMJ ?= { var:tot_avmj_nrhd_claim > 0 } c:NRHD ?= this }
		}
	}
	should_show_when_not_involved = {
		OR = {
			c:AVMJ ?= this
			AND = { c:AVMJ ?= { var:tot_avmj_htt_claim > 0 } c:HTT ?= this }
			AND = { c:AVMJ ?= { var:tot_avmj_gbdp_claim > 0 } c:GBDP ?= this }
			AND = { c:AVMJ ?= { var:tot_avmj_mygo_claim > 0 var:tot_avmj_mygo_claim_holder = 0 } c:MYG ?= this }
			AND = { c:AVMJ ?= { var:tot_avmj_nrhd_claim > 0 } c:NRHD ?= this }
		}
	}
	possible = {
		OR = {
			c:AVMJ ?= this
			AND = { c:AVMJ ?= { var:tot_avmj_htt_claim > 0 } c:HTT ?= this }
			AND = { c:AVMJ ?= { var:tot_avmj_gbdp_claim > 0 } c:GBDP ?= this }
			AND = { c:AVMJ ?= { var:tot_avmj_mygo_claim > 0 var:tot_avmj_mygo_claim_holder = 0 } c:MYG ?= this }
			AND = { c:AVMJ ?= { var:tot_avmj_nrhd_claim > 0 } c:NRHD ?= this }
		}
	}

	immediate = {
		tot_avmj_setup_debt = yes
	}

	complete = {
		custom_tooltip = {
			text = je_tot_avmj_debt_crisis_complete_tt
			var:tot_avmj_debt_principal <= 0
		}
	}

	on_complete = {
		remove_modifier = tot_avmj_debt_burden
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_tot_avmj_debt_crisis_status_chamber
				trigger = { var:tot_avmj_mygo_claim_holder = 1 }
			}
			triggered_desc = {
				desc = je_tot_avmj_debt_crisis_status
				trigger = { always = yes }
			}
		}
	}
}
