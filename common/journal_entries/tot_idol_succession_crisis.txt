je_tot_idol_succession_crisis = {
	icon = "gfx/interface/icons/event_icons/event_diplomacy.dds"
	group = je_group_tot_idol_succession
	should_be_pinned_by_default = yes
	can_revolution_inherit = yes
	weight = 10000
	timeout = 3650
	reason = je_tot_idol_succession_crisis_reason
	on_timeout = {
		if = {
			limit = { c:MUS ?= this }
			trigger_event = tot_idol_succession.2
		}
	}

	is_shown_in_lobby = {
		OR = {
			c:MUS ?= this
			c:AQO ?= this
			c:NJS ?= this
			c:IDMS ?= this
			c:GKMS ?= this
			c:GBDP ?= this
			c:HTT ?= this
			c:PRU ?= this
		}
	}

	should_be_involved = {
		OR = {
			c:MUS ?= this
			c:AQO ?= this
			c:NJS ?= this
			c:IDMS ?= this
			c:GKMS ?= this
			c:GBDP ?= this
			c:HTT ?= this
			c:PRU ?= this
		}
	}

	should_show_when_not_involved = {
		always = no
	}

	is_shown_when_inactive = {
		OR = {
			c:MUS ?= this
			c:AQO ?= this
			c:NJS ?= this
			c:IDMS ?= this
			c:GKMS ?= this
			c:GBDP ?= this
			c:HTT ?= this
			c:PRU ?= this
		}
	}

	possible = {
		OR = {
			c:MUS ?= this
			c:AQO ?= this
			c:NJS ?= this
			c:IDMS ?= this
			c:GKMS ?= this
			c:GBDP ?= this
			c:HTT ?= this
			c:PRU ?= this
		}
	}

	immediate = {
		add_journal_entry = { type = je_tot_golden_hall_project }
		c:MUS ?= {
			if = {
				limit = { NOT = { has_journal_entry = je_tot_ll_publication_program } }
				add_journal_entry = { type = je_tot_ll_publication_program }
			}
		}
		c:IDMS ?= {
			if = {
				limit = { NOT = { has_journal_entry = je_tot_imas_publication_program } }
				add_journal_entry = { type = je_tot_imas_publication_program }
			}
		}
		scope:journal_entry = { add_involved_country = c:MUS }
		scope:journal_entry = { add_involved_country = c:AQO }
		scope:journal_entry = { add_involved_country = c:NJS }
		scope:journal_entry = { add_involved_country = c:IDMS }
		scope:journal_entry = { add_involved_country = c:GKMS }
		scope:journal_entry = { add_involved_country = c:GBDP }
		scope:journal_entry = { add_involved_country = c:HTT }
		scope:journal_entry = { add_involved_country = c:PRU }
		every_country = {
			limit = { tot_golden_hall_crisis_participant_trigger = yes }
			set_variable = { name = tot_golden_hall_crisis_participant value = 1 }
			set_variable = { name = tot_idol_crisis_ll_score value = 0 }
			set_variable = { name = tot_idol_crisis_imas_score value = 0 }
			set_variable = { name = tot_idol_crisis_gbdp_score value = 0 }
			set_variable = { name = tot_idol_crisis_ger_score value = 0 }
			set_variable = { name = tot_idol_crisis_htt_score value = 0 }
			set_variable = { name = tot_idol_crisis_tension value = 20 }
			set_variable = { name = tot_idol_crisis_compromise_score value = 30 }
			set_variable = { name = tot_golden_hall_progress value = 0 }
		}
		tot_golden_hall_create_site = yes
		scope:journal_entry = {
			set_bar_progress = {
				name = je_tot_idol_crisis_tension_bar
				value = root.var:tot_idol_crisis_tension
			}
			set_bar_progress = {
				name = je_tot_idol_crisis_compromise_bar
				value = root.var:tot_idol_crisis_compromise_score
			}
		}
	}

	scripted_progress_bar = je_tot_idol_crisis_tension_bar
	scripted_progress_bar = je_tot_idol_crisis_compromise_bar

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = je_tot_idol_succession_crisis_status
				trigger = { always = yes }
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			every_country = {
				limit = { tot_golden_hall_crisis_participant_trigger = yes }
				tot_golden_hall_process_contracts = yes
			}
			scope:journal_entry = {
				set_bar_progress = {
					name = je_tot_idol_crisis_tension_bar
					value = root.var:tot_idol_crisis_tension
				}
				set_bar_progress = {
					name = je_tot_idol_crisis_compromise_bar
					value = root.var:tot_idol_crisis_compromise_score
				}
			}
			if = {
				limit = {
					c:MUS ?= this
					var:tot_idol_crisis_tension >= 100
				}
				trigger_event = tot_idol_succession.1
			}
		}
	}

	complete = {
		custom_tooltip = {
			text = je_tot_idol_succession_crisis_complete_tt
			always = no
		}
	}

	goal_add_value = {
		add = 100
	}

	should_be_pinned_by_default_uninvolved_or_context = yes
}
