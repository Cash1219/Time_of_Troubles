# Music competition scoring values.
# The game keeps the visible score as an integer, while the raw performance
# and multiplier are stored separately for the activity panel and rewards.

tot_music_competition_band_1_multiplier_value = {
	value = 1
	if = { limit = { has_variable = tot_band_1_creation_progress } add = { value = band_1_kizuna_current_level_value multiply = 0.4 } }
	if = { limit = { has_variable = tot_band_1_member_count var:tot_band_1_member_count >= 2 } add = 0.25 }
	if = { limit = { has_variable = tot_band_1_member_count var:tot_band_1_member_count >= 3 } add = 0.25 }
	if = { limit = { has_variable = tot_band_1_member_count var:tot_band_1_member_count >= 4 } add = 0.25 }
	if = { limit = { has_variable = tot_band_1_member_count var:tot_band_1_member_count >= 5 } add = 0.25 }
	if = { limit = { has_variable = tot_band_1_member_count var:tot_band_1_member_count >= 6 } add = 0.25 }
	if = { limit = { has_variable = tot_music_competition_band_1_mastery_bonus } add = var:tot_music_competition_band_1_mastery_bonus }
	if = { limit = { has_variable = tot_music_competition_event_modifier } add = var:tot_music_competition_event_modifier }
	max = 5
}

# Idol events deliberately use a separate progression track from bands.  An
# idol's own level and idol-specific traits matter in the solo event; a group
# gains strength from the number of prepared historical idols it can field.
tot_music_competition_idol_solo_multiplier_value = {
	value = 1
	if = {
		limit = { has_variable = tot_music_competition_idol_representative }
		var:tot_music_competition_idol_representative ?= {
			if = { limit = { has_trait = tot_group_sheet_023_trait } add = 0.9 }
			if = { limit = { has_trait = tot_group_sheet_060_trait } add = 0.6 }
			if = { limit = { has_trait = tot_group_sheet_056_trait } add = 0.2 }
			if = {
				limit = { has_variable = tot_character_growth_level }
				add = { value = var:tot_character_growth_level multiply = 0.4 }
			}
		}
	}
	if = { limit = { has_variable = tot_music_competition_event_modifier } add = var:tot_music_competition_event_modifier }
	max = 7
}
