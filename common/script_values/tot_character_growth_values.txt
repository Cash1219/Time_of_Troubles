tot_character_growth_current_cost_value = {
	value = 1000

	if = {
		limit = { has_variable = tot_character_growth_trait_cost_mult }
		multiply = 100
		add = var:tot_character_growth_trait_cost_mult
		divide = 100
	}
	if = {
		limit = { has_variable = tot_character_growth_club_cost_mult }
		multiply = 100
		add = var:tot_character_growth_club_cost_mult
		divide = 100
	}

	if = {
		limit = { has_variable = tot_character_growth_trait_cost_add }
		add = var:tot_character_growth_trait_cost_add
	}

	if = {
		limit = { tot_character_growth_acquired_traits_count_value > 0 }
		add = {
			value = tot_character_growth_acquired_traits_count_value
			multiply = 3000
		}
	}

	min = 1
}
