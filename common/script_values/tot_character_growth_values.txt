tot_character_growth_exp_progress_value = {
	value = 0

	if = {
		limit = {
			has_variable = tot_character_growth_exp
		}
		add = var:tot_character_growth_exp
	}

	divide = 1000
	min = 0
	max = 1
}

tot_character_growth_weekly_gain_value = {
	value = 10

	if = {
		limit = {
			is_ruler_of_own_country = yes
		}
		add = 10
	}

	if = {
		limit = {
			interest_group ?= {
				is_interest_group_type = ig_landowners
			}
		}
		add = 5
	}
}
