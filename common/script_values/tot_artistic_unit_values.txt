tot_special_unit_ratio = {
	value = modifier:country_tot_special_unit_ratio_add
	min = 0
	max = 1
}

tot_idol_guard_unit_ratio = {
	value = modifier:country_tot_idol_guard_unit_ratio_add
	min = 0
	max = 1
}

tot_idol_knight_unit_ratio = {
	value = modifier:country_tot_idol_knight_unit_ratio_add
	min = 0
	max = 1
}

tot_special_units_total_army_units = {
	value = 0
	every_military_formation = {
		limit = { is_army = yes }
		add = num_units
	}
}

tot_idol_guard_unit_capacity = {
	value = tot_special_units_total_army_units
	multiply = tot_idol_guard_unit_ratio
	round = yes
	min = 0
}

tot_idol_guard_unit_free_capacity = {
	value = tot_idol_guard_unit_capacity
	subtract = tot_idol_guard_unit_deployed
	min = 0
}

tot_idol_knight_unit_capacity = {
	value = tot_special_units_total_army_units
	multiply = tot_idol_knight_unit_ratio
	round = yes
	min = 0
}

tot_idol_knight_unit_free_capacity = {
	value = tot_idol_knight_unit_capacity
	subtract = tot_idol_knight_unit_deployed
	min = 0
}

tot_special_unit_deployed = {
	value = {
		every_military_formation = {
			limit = { is_army = yes }
			every_combat_unit = {
				limit = {
					OR = {
						has_unit_type = unit_type:combat_unit_type_tot_idol_guard
						has_unit_type = unit_type:combat_unit_type_tot_idol_knight
					}
				}
				add = 1
			}
		}
	}
}

tot_idol_guard_unit_deployed = {
	value = {
		every_military_formation = {
			limit = { is_army = yes }
			every_combat_unit = {
				limit = { has_unit_type = unit_type:combat_unit_type_tot_idol_guard }

		}
	}
}


tot_idol_knight_unit_deployed = {

}
