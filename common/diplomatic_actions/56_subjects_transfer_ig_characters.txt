transfer_ig_landowners_characters = {
	groups = { general }
	texture = "gfx/interface/icons/diplomatic_action_icons/da_appoint_colonial_governor.dds"
	requires_approval = yes
	should_notify_third_parties = yes
	show_effect_in_tooltip = yes
	can_use_obligations = no
	potential = {
		scope:target_country = {
			NOT = { is_country_type = decentralized }
			NOT = { this = root }
		}
	}
	possible = {
		scope:target_country = {
			NOT = { is_country_type = decentralized }
			NOT = { this = root }
			any_scope_character = {
				interest_group ?= { is_interest_group_type = ig_landowners }
			}
		}
	}
	accept_effect = {
		scope:target_country = {
			every_scope_character = {
				limit = { interest_group ?= { is_interest_group_type = ig_landowners } }
				transfer_character = root
			}
		}
	}
	ai = {
		evaluation_chance = { value = 0.1 }
		propose_score = { value = 10 }
		accept_score = {
			add = { desc = DIPLOMATIC_ACCEPTANCE_BASE value = -50 }
			if = {
				limit = { scope:target_country = { is_subject_of = root } }
				add = { desc = DIPLOMATIC_ACCEPTANCE_TRANSFER_SUBJECT_BONUS value = 30 }
			}
		}
	}
}
