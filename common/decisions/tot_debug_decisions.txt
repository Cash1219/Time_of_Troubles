			}
			add_journal_entry = { type = je_tot_character_growth_gui }

		if = {
			limit = {
				NOT = { has_journal_entry = je_kizuna_gui }
			}

			limit = {

			}

		}
		tot_character_growth_refresh = yes
		update_kizuna = yes
	}

		value = 0
	}
}
