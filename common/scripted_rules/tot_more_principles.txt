# Minimal support for the additional principle slots shown by the ToT power bloc GUI.

unlock_power_bloc_principle_slot_5 = {
	hidden_trigger = {
		has_dlc_feature = power_bloc_features
	}
	OR = {
		hidden_trigger = {
			used_principle_slots >= 5
		}
		AND = {
			power_bloc_rank <= 5
			num_power_bloc_members >= 5
		}
	}
}
