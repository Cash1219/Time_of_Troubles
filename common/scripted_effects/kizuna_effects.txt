tot_gourmet_check_rewards = {
	if = {
		limit = {
			var:tot_gourmet_progress >= 3500
			NOT = { has_variable = tot_gourmet_tier_1_claimed }
		}
		set_variable = { name = tot_gourmet_tier_1_claimed value = 1 }
		add_modifier = tot_gourmet_reward_tier_1_modifier
	}
	if = {
		limit = {
			var:tot_gourmet_progress >= 8500
			NOT = { has_variable = tot_gourmet_tier_2_claimed }
		}
		set_variable = { name = tot_gourmet_tier_2_claimed value = 1 }
		add_modifier = tot_gourmet_reward_tier_2_modifier
	}
	if = {
		limit = {
			var:tot_gourmet_progress >= 14000
			NOT = { has_variable = tot_gourmet_tier_3_claimed }
		}
		set_variable = { name = tot_gourmet_tier_3_claimed value = 1 }
		add_modifier = tot_gourmet_reward_tier_3_modifier
	}
	if = {
		limit = {
			var:tot_gourmet_progress >= 22000
			NOT = { has_variable = tot_gourmet_tier_4_claimed }
		}
		set_variable = { name = tot_gourmet_tier_4_claimed value = 1 }
		add_modifier = tot_gourmet_reward_tier_4_modifier
	}
	if = {
		limit = {
			var:tot_gourmet_progress >= 36000
			NOT = { has_variable = tot_gourmet_tier_5_claimed }
		}
		set_variable = { name = tot_gourmet_tier_5_claimed value = 1 }
		add_modifier = tot_gourmet_reward_tier_5_modifier
	}
}
