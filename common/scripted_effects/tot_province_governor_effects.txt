tot_update_province_governor_candidates = {
	if = {
		limit = {
			has_variable_list = tot_province_governor_candidates
		}
		clear_variable_list = tot_province_governor_candidates
	}

	every_scope_character = {
		limit = {
			is_character_alive = yes
			NOT = {
				has_role = character_role_tot_province_governor
			}
			OR = {
				owner = {
					NOT = { has_variable = tot_governor_candidate_duty_filter }
				}
				owner = {
					var:tot_governor_candidate_duty_filter = 0
				}
				AND = {
					owner = {
						var:tot_governor_candidate_duty_filter = 1
					}
					OR = {
						has_role = character_role_general
						has_role = character_role_admiral
						has_role = character_role_executive
						has_role = character_role_magnate
						has_role = character_role_agitator
						has_role = character_role_tot_student_council_president
						has_role = character_role_tot_discipline_minister
						has_role = character_role_tot_education_minister
						has_role = character_role_tot_propaganda_minister
						has_role = character_role_tot_chief_of_staff
						has_role = character_role_tot_honorary_official
					}
				}
				AND = {
					owner = {
						var:tot_governor_candidate_duty_filter = 2
					}
					NOT = {
						OR = {
							has_role = character_role_general
							has_role = character_role_admiral
							has_role = character_role_executive
							has_role = character_role_magnate
							has_role = character_role_agitator
							has_role = character_role_tot_student_council_president
							has_role = character_role_tot_discipline_minister
							has_role = character_role_tot_education_minister
							has_role = character_role_tot_propaganda_minister
							has_role = character_role_tot_chief_of_staff
							has_role = character_role_tot_honorary_official
						}
					}
				}
			}
			# Keep the governor page's filters separate from the minister candidate filters.
			OR = {
				owner = {
					NOT = { has_variable = tot_governor_candidate_filter }
				}
				owner = {
					var:tot_governor_candidate_filter = 0
				}
				AND = {
					owner = {
						var:tot_governor_candidate_filter = 1
					}
					is_historical = yes
				}
				AND = {
					owner = {
						var:tot_governor_candidate_filter = 2
					}
					is_historical = no
				}
			}
		}
		owner = {
			add_to_variable_list = {
				name = tot_province_governor_candidates
				target = prev
			}
		}
	}
}

tot_character_growth_refresh = {
	tot_update_province_governor_candidates = yes

	# Migrate existing governors into the new classification system on the next UI refresh.
	every_scope_state = {
		limit = {
			owner ?= { is_player = yes }
			has_variable = tot_province_governor
		}
		tot_apply_province_governor_type_modifier = yes
	}

	if = {
		limit = {
			has_variable_list = tot_character_growth_characters
		}
		clear_variable_list = tot_character_growth_characters
	}

	every_scope_character = {
		limit = {
			is_character_alive = yes
			historical = yes
		}
		if = {
			limit = {
				NOT = { has_variable = tot_character_growth_exp }
			}
			set_variable = {
				name = tot_character_growth_exp
				value = 0
			}
		}
		if = {
			limit = { NOT = { has_variable = tot_character_growth_level } }
			# Old saves used gained-trait count as the implicit level, so retain their current progress tier.
			set_variable = { name = tot_character_growth_level value = tot_character_growth_acquired_traits_count_value }
		}
		if = {
			limit = { NOT = { has_variable = tot_character_growth_skill_points } }
			set_variable = { name = tot_character_growth_skill_points value = 0 }
		}
		owner = {
			add_to_variable_list = {
				name = tot_character_growth_characters
				target = prev
			}
		}
	}

	if = {
		limit = {
			has_variable_list = tot_character_growth_governor_states
		}
		clear_variable_list = tot_character_growth_governor_states
	}

	every_scope_state = {
		limit = {
			owner ?= {
				is_player = yes
			}
		}
		owner = {
			add_to_variable_list = {
				name = tot_character_growth_governor_states
				target = prev
			}
		}
	}

	if = {
		limit = {
			has_variable_list = tot_character_growth_selected_governor_state_list
		}
		clear_variable_list = tot_character_growth_selected_governor_state_list
	}

	if = {
		limit = {
			has_variable = tot_character_growth_selected_governor_state
		}
		var:tot_character_growth_selected_governor_state ?= {
			owner = {
				add_to_variable_list = {
					name = tot_character_growth_selected_governor_state_list
					target = prev
				}
			}
		}
	}
}

tot_clear_state_governor = {
	if = {
		limit = {
			has_variable = tot_province_governor
		}
		var:tot_province_governor ?= {
			remove_variable = tot_governor_state
			remove_character_role = character_role_tot_province_governor
		}
		remove_variable = tot_province_governor
	}

	if = {
		limit = {
			has_modifier = tot_province_governor_modifier
		}
		remove_modifier = tot_province_governor_modifier
	}

	remove_modifier = tot_province_governor_administration_modifier
	remove_modifier = tot_province_governor_finance_modifier
	remove_modifier = tot_province_governor_industry_modifier
	remove_modifier = tot_province_governor_education_modifier
	remove_modifier = tot_province_governor_military_modifier
	remove_variable = tot_province_governor_type

	owner ?= {
		tot_update_province_governor_candidates = yes
	}
}

# Each province has one governor and one selected administrative direction.
tot_apply_province_governor_type_modifier = {
	remove_modifier = tot_province_governor_administration_modifier
	remove_modifier = tot_province_governor_finance_modifier
	remove_modifier = tot_province_governor_industry_modifier
	remove_modifier = tot_province_governor_education_modifier
	remove_modifier = tot_province_governor_military_modifier

	if = {
		limit = { has_variable = tot_province_governor }
		if = {
			limit = { NOT = { has_variable = tot_province_governor_type } }
			set_variable = { name = tot_province_governor_type value = 0 }
		}
		if = { limit = { var:tot_province_governor_type = 0 } add_modifier = { name = tot_province_governor_administration_modifier } }
		if = { limit = { var:tot_province_governor_type = 1 } add_modifier = { name = tot_province_governor_finance_modifier } }
		if = { limit = { var:tot_province_governor_type = 2 } add_modifier = { name = tot_province_governor_industry_modifier } }
		if = { limit = { var:tot_province_governor_type = 3 } add_modifier = { name = tot_province_governor_education_modifier } }
		if = { limit = { var:tot_province_governor_type = 4 } add_modifier = { name = tot_province_governor_military_modifier } }
	}
}

tot_appoint_province_governor_effect = {
	if = {
		limit = {
			is_character_alive = yes
			NOT = {
				has_role = character_role_tot_province_governor
			}
			owner ?= {
				tot_province_governor_limit > tot_province_governor_count
			}
			scope:tot_governor_state ?= {
				owner ?= {
					is_player = yes
				}
				NOT = {
					has_variable = tot_province_governor
				}
				NOT = {
					has_variable = tot_province_governor_replace_cooldown
				}
			}
		}

		save_temporary_scope_as = tot_selected_governor

		if = {
			limit = {
				has_variable = tot_governor_state
			}
			var:tot_governor_state ?= {
				tot_clear_state_governor = yes
			}
		}

		scope:tot_governor_state ?= {
			tot_clear_state_governor = yes
			set_variable = {
				name = tot_province_governor
				value = scope:tot_selected_governor
			}
			add_modifier = {
				name = tot_province_governor_modifier
			}
			tot_apply_province_governor_type_modifier = yes
			set_variable = {
				name = tot_province_governor_replace_cooldown
				years = 10
			}
		}

		add_character_role = character_role_tot_province_governor
		set_variable = {
			name = tot_governor_state
			value = scope:tot_governor_state
		}

		owner ?= {
			tot_update_province_governor_candidates = yes
		}
	}
}

tot_appoint_selected_province_governor_effect = {
	if = {
		limit = {
			is_character_alive = yes
			NOT = {
				has_role = character_role_tot_province_governor
			}
			owner ?= {
				has_variable = tot_character_growth_selected_governor_state
				tot_province_governor_limit > tot_province_governor_count
				var:tot_character_growth_selected_governor_state ?= {
					owner ?= {
						is_player = yes
					}
					NOT = {
						has_variable = tot_province_governor
					}
					NOT = {
						has_variable = tot_province_governor_replace_cooldown
					}
				}
			}
		}

		save_temporary_scope_as = tot_selected_governor

		owner = {
			var:tot_character_growth_selected_governor_state ?= {
				save_temporary_scope_as = tot_selected_governor_state
			}
		}

		if = {
			limit = {
				has_variable = tot_governor_state
			}
			var:tot_governor_state ?= {
				tot_clear_state_governor = yes
			}
		}

		scope:tot_selected_governor_state ?= {
			tot_clear_state_governor = yes
			set_variable = {
				name = tot_province_governor
				value = scope:tot_selected_governor
			}
			add_modifier = {
				name = tot_province_governor_modifier
			}
			tot_apply_province_governor_type_modifier = yes
			set_variable = {
				name = tot_province_governor_replace_cooldown
				years = 10
			}
		}

		add_character_role = character_role_tot_province_governor
		set_variable = {
			name = tot_governor_state
			value = scope:tot_selected_governor_state
		}

		owner ?= {
			tot_character_growth_refresh = yes
		}
	}
}
