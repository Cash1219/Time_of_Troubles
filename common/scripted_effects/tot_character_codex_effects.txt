tot_character_codex_refresh_bang_dream_effect = {
	# Preserve the player scope while walking the global character pool.
	save_temporary_scope_as = tot_character_codex_owner
	set_variable = { name = tot_character_codex_selection value = 8 }
	set_variable = { name = tot_character_codex_result_count value = 0 }
	if = {
		limit = { has_variable_list = tot_character_codex_results }
		clear_variable_list = tot_character_codex_results
	}

	# This deliberately runs only from the player's manual GUI action. Do not add it
	# to a pulse: the codex is a cached world lookup rather than a live game system.
	# Characters are scoped through their owning country in Victoria 3.  Keep the
	# player list as the destination while every_country walks the world pool.
	every_country = {
		every_scope_character = {
			limit = {
				is_character_alive = yes
				is_in_void = no
				is_in_exile_pool = no
				has_trait = tot_group_sheet_008_trait
			}
			save_scope_as = tot_character_codex_entry
			scope:tot_character_codex_owner = {
				add_to_variable_list = {
					name = tot_character_codex_results
					target = scope:tot_character_codex_entry
				}
				change_variable = { name = tot_character_codex_result_count add = 1 }
			}
		}
	}

	# Keep the page useful even on saves where the global country iterator has
	# not populated character scopes yet. The player's own roster is a safe
	# fallback and will be replaced by the full world scan on the next refresh.
	if = {
		limit = { var:tot_character_codex_result_count = 0 }
		every_scope_character = {
			limit = {
				is_character_alive = yes
				is_in_void = no
				is_in_exile_pool = no
				has_trait = tot_group_sheet_008_trait
			}
			save_scope_as = tot_character_codex_entry
			add_to_variable_list = {
				name = tot_character_codex_results
				target = scope:tot_character_codex_entry
			}
			change_variable = { name = tot_character_codex_result_count add = 1 }
		}
	}

	update_kizuna = yes
}
