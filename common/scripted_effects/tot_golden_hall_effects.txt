			limit = { tot_golden_hall_crisis_participant_trigger = yes }
			change_variable = { name = tot_idol_crisis_ll_score add = $VALUE$ }
		}
	}
	if = {
		limit = { var:tot_golden_hall_contract_faction = 2 }
		every_country = {
			limit = { tot_golden_hall_crisis_participant_trigger = yes }
			change_variable = { name = tot_idol_crisis_imas_score add = $VALUE$ }
		}
	}
	if = {
		limit = { var:tot_golden_hall_contract_faction = 3 }
		every_country = {
			limit = { tot_golden_hall_crisis_participant_trigger = yes }
			change_variable = { name = tot_idol_crisis_gbdp_score add = $VALUE$ }
		}
	}
	if = {
		limit = { var:tot_golden_hall_contract_faction = 4 }
		every_country = {
			limit = { tot_golden_hall_crisis_participant_trigger = yes }
			change_variable = { name = tot_idol_crisis_htt_score add = $VALUE$ }
		}
	}
	if = {
		limit = { var:tot_golden_hall_contract_faction = 5 }
		every_country = {
			limit = { tot_golden_hall_crisis_participant_trigger = yes }
			change_variable = { name = tot_idol_crisis_ger_score add = $VALUE$ }
		}
	}
}

tot_golden_hall_add_progress = {
	s:STATE_AUSTRIA = {
		add_building_level = {
			building = building_tot_golden_hall_construction_site
			level = $PROGRESS$
		}
		owner = { change_variable = { name = tot_golden_hall_progress add = $PROGRESS$ } }
	}
	if = {
		limit = { has_variable = tot_golden_hall_crisis_participant }
		tot_golden_hall_add_crisis_score = { VALUE = $SCORE$ }
		every_country = {
			limit = { tot_golden_hall_crisis_participant_trigger = yes }
			change_variable = { name = tot_golden_hall_progress add = $PROGRESS$ }
		}
	}
	tot_golden_hall_check_completion = yes
}

tot_golden_hall_set_contractor_faction = {
	if = { limit = { OR = { c:MUS ?= this c:AQO ?= this c:NJS ?= this } } set_variable = { name = tot_golden_hall_contract_faction value = 1 } }
	if = { limit = { OR = { c:IDMS ?= this c:GKMS ?= this } } set_variable = { name = tot_golden_hall_contract_faction value = 2 } }
	if = { limit = { c:GBDP ?= this } set_variable = { name = tot_golden_hall_contract_faction value = 3 } }
	if = { limit = { c:HTT ?= this } set_variable = { name = tot_golden_hall_contract_faction value = 4 } }
	if = { limit = { c:PRU ?= this } set_variable = { name = tot_golden_hall_contract_faction value = 5 } }
}

tot_golden_hall_mark_naming_complete = {
	if = {
		limit = { var:tot_golden_hall_contract_faction = 1 }
		every_country = { limit = { OR = { c:MUS ?= this c:AQO ?= this c:NJS ?= this } } set_variable = { name = tot_golden_hall_naming_done value = 1 } }
	}
	if = {
		limit = { var:tot_golden_hall_contract_faction = 2 }
		every_country = { limit = { OR = { c:IDMS ?= this c:GKMS ?= this } } set_variable = { name = tot_golden_hall_naming_done value = 1 } }
	}
	if = { limit = { var:tot_golden_hall_contract_faction = 3 } set_variable = { name = tot_golden_hall_naming_done value = 1 } }
	if = { limit = { var:tot_golden_hall_contract_faction = 4 } set_variable = { name = tot_golden_hall_naming_done value = 1 } }
	if = { limit = { var:tot_golden_hall_contract_faction = 5 } set_variable = { name = tot_golden_hall_naming_done value = 1 } }
}

tot_golden_hall_process_contracts = {
	# This effect is called from a journal entry's country scope. Preserve the
	# country currently being checked so a capital-state loop cannot fall back
	# to the journal holder and credit its faction for another country's work.
	save_temporary_scope_as = tot_golden_hall_contractor
	every_scope_state = {
		limit = { is_capital = yes }
		if = {
			limit = { has_building = building_tot_golden_hall_small_contract }
			remove_building = building_tot_golden_hall_small_contract
			scope:tot_golden_hall_contractor = {
				tot_golden_hall_set_contractor_faction = yes
				tot_golden_hall_add_progress = { PROGRESS = 1 SCORE = 1 }
				remove_variable = tot_golden_hall_contract_faction
			}
		}
		if = {
			limit = { has_building = building_tot_golden_hall_large_contract }
			remove_building = building_tot_golden_hall_large_contract
			scope:tot_golden_hall_contractor = {
				tot_golden_hall_set_contractor_faction = yes
				tot_golden_hall_add_progress = { PROGRESS = 2 SCORE = 3 }
				add_modifier = { name = modifier_tot_golden_hall_stage_expansion years = 5 }
				remove_variable = tot_golden_hall_contract_faction
			}
		}
		if = {
			limit = { has_building = building_tot_golden_hall_naming_contract }
			remove_building = building_tot_golden_hall_naming_contract
			scope:tot_golden_hall_contractor = {
				if = {
					limit = { NOT = { has_variable = tot_golden_hall_naming_done } }
					tot_golden_hall_set_contractor_faction = yes
					tot_golden_hall_add_progress = { PROGRESS = 4 SCORE = 7 }
					tot_golden_hall_mark_naming_complete = yes
					add_modifier = { name = modifier_tot_golden_hall_naming_prestige years = 10 }
					remove_variable = tot_golden_hall_contract_faction
				}
			}
		}
	}
}

tot_golden_hall_check_completion = {
	if = {
		limit = { NOT = { has_variable = tot_golden_hall_built } }
		s:STATE_AUSTRIA = {
			if = {
				limit = {
					any_scope_building = {
						is_building_type = building_tot_golden_hall_construction_site
						level >= 31
					}
				}
				remove_building = building_tot_golden_hall_construction_site
				if = {
					limit = { NOT = { has_building = building_tot_golden_hall_monument } }
					create_building = {
						building = building_tot_golden_hall_monument
						level = 1
						reserves = 1
					}
				}
				add_building_level = {
					building = building_stage
					level = 5
				}
				owner = {
					set_variable = { name = tot_golden_hall_built value = 1 }
					add_modifier = { name = modifier_tot_golden_hall_complete_owner years = 10 }
				}
				every_country = {
					limit = { tot_golden_hall_crisis_participant_trigger = yes }
					set_variable = { name = tot_golden_hall_built value = 1 }
					change_variable = { name = tot_idol_crisis_tension add = -35 }
					change_variable = { name = tot_idol_crisis_compromise_score add = 15 }
				}
			}
		}
	}
}

tot_golden_hall_create_site = {
	s:STATE_AUSTRIA = {
		if = {
			limit = {
				NOT = { has_building = building_tot_golden_hall_construction_site }
				NOT = { has_building = building_tot_golden_hall_monument }
			}
			create_building = {
				building = building_tot_golden_hall_construction_site
				level = 1
				reserves = 1
			}
		}
	}
}

tot_golden_hall_handoff_to_vienna_owner = {
	if = {
		limit = { NOT = { has_variable = tot_golden_hall_built } }
		s:STATE_AUSTRIA = {
			if = {
				limit = { has_building = building_tot_golden_hall_construction_site }
				owner = {
					if = {
						limit = { NOT = { has_journal_entry = je_tot_golden_hall_continuation } }
						add_journal_entry = { type = je_tot_golden_hall_continuation }
					}
				}
			}
		}
	}
}
