tot_avmj_transfer_mygo_claim = {
	if = {
		limit = { var:tot_avmj_mygo_claim_holder = 0 }
		set_variable = { name = tot_avmj_mygo_claim_holder value = 1 }
		add_modifier = { name = tot_avmj_aleppo_chamber }
		# The company is the visible local successor for the former MYGO claim.
		add_company = company_type:tot_company_aleppo_chamber
		company:tot_company_aleppo_chamber = {
			set_company_establishment_date = 1836.1.1
			set_company_state_region = s:STATE_ALEPPO
		}
	}
}

tot_avmj_update_creditor_holders = {
	c:HTT = { save_scope_as = tot_avmj_htt_country }
	c:GBDP = { save_scope_as = tot_avmj_gbdp_country }
	c:MYG = { save_scope_as = tot_avmj_mygo_country }
	c:NRHD = { save_scope_as = tot_avmj_nrhd_country }
	tot_avmj_refresh_creditor_gui_lists = yes
	if = {
		limit = { NOT = { c:MYG ?= { is_country_alive = yes } } var:tot_avmj_mygo_claim_holder = 0 }
		tot_avmj_transfer_mygo_claim = yes
	}
	if = { limit = { NOT = { c:HTT ?= { is_country_alive = yes } } var:tot_avmj_htt_claim_holder = 0 } set_variable = { name = tot_avmj_htt_claim_holder value = 1 } }

# The selected level is collected once per month. A payment is added to the
# fund only in the same branch that removes it from AVMJ's treasury.
tot_avmj_collect_monthly_repayment = {
	set_variable = { name = tot_avmj_last_monthly_repayment value = 0 }
	if = {
		limit = { var:tot_avmj_repayment_level = 1 gold_reserves >= 1000 }
		add_treasury = -1000
		change_variable = { name = tot_avmj_debt_fund add = 1000 }
		set_variable = { name = tot_avmj_last_monthly_repayment value = 1000 }
	}
	else_if = {
		limit = { var:tot_avmj_repayment_level = 2 gold_reserves >= 2500 }
		add_treasury = -2500
		change_variable = { name = tot_avmj_debt_fund add = 2500 }
		set_variable = { name = tot_avmj_last_monthly_repayment value = 2500 }
	}
	else_if = {
		limit = { var:tot_avmj_repayment_level = 3 gold_reserves >= 5000 }
		add_treasury = -5000
		change_variable = { name = tot_avmj_debt_fund add = 5000 }
		set_variable = { name = tot_avmj_last_monthly_repayment value = 5000 }
	}
	else_if = {
		limit = { var:tot_avmj_repayment_level = 4 gold_reserves >= 10000 }
		add_treasury = -10000
		change_variable = { name = tot_avmj_debt_fund add = 10000 }
		set_variable = { name = tot_avmj_last_monthly_repayment value = 10000 }
	}
	else_if = {
		limit = { var:tot_avmj_repayment_level = 5 gold_reserves >= 20000 }
		add_treasury = -20000
		change_variable = { name = tot_avmj_debt_fund add = 20000 }
		set_variable = { name = tot_avmj_last_monthly_repayment value = 20000 }
	}
	if = { limit = { var:tot_avmj_repayment_level = 1 } add_modifier = { name = tot_avmj_debt_low_repayment months = 2 } }
	if = { limit = { var:tot_avmj_repayment_level = 4 } add_modifier = { name = tot_avmj_debt_high_repayment months = 2 } }
	if = { limit = { var:tot_avmj_repayment_level = 5 } add_modifier = { name = tot_avmj_debt_maximum_repayment months = 2 } }
}

tot_avmj_settle_assembly = {
	set_variable = { name = tot_avmj_expected_interest value = var:tot_avmj_interest_due }
	set_variable = { name = tot_avmj_last_assembly_fund value = var:tot_avmj_debt_fund }
	set_variable = { name = tot_avmj_actual_interest_payment value = var:tot_avmj_debt_fund }
	if = { limit = { var:tot_avmj_actual_interest_payment > var:tot_avmj_interest_due } set_variable = { name = tot_avmj_actual_interest_payment value = var:tot_avmj_interest_due } }
	set_variable = { name = tot_avmj_last_assembly_interest value = var:tot_avmj_actual_interest_payment }
	set_variable = { name = tot_avmj_last_assembly_shortfall value = var:tot_avmj_interest_due }
	change_variable = { name = tot_avmj_last_assembly_shortfall add = { value = var:tot_avmj_actual_interest_payment multiply = -1 } }
	change_variable = { name = tot_avmj_debt_fund add = { value = var:tot_avmj_actual_interest_payment multiply = -1 } }
	change_variable = { name = tot_avmj_interest_due add = { value = var:tot_avmj_actual_interest_payment multiply = -1 } }
	if = { limit = { var:tot_avmj_actual_interest_payment > 0 } tot_avmj_distribute_interest = yes }

	set_variable = { name = tot_avmj_actual_principal_payment value = var:tot_avmj_debt_fund }
	if = { limit = { var:tot_avmj_actual_principal_payment > var:tot_avmj_debt_principal } set_variable = { name = tot_avmj_actual_principal_payment value = var:tot_avmj_debt_principal } }
	set_variable = { name = tot_avmj_last_assembly_principal value = var:tot_avmj_actual_principal_payment }
	change_variable = { name = tot_avmj_debt_fund add = { value = var:tot_avmj_actual_principal_payment multiply = -1 } }
	change_variable = { name = tot_avmj_debt_principal add = { value = var:tot_avmj_actual_principal_payment multiply = -1 } }
	if = { limit = { var:tot_avmj_actual_principal_payment > 0 } tot_avmj_distribute_principal = yes }

	set_variable = { name = tot_avmj_last_assembly_surplus value = var:tot_avmj_debt_fund }
	if = { limit = { var:tot_avmj_last_assembly_shortfall > 0 } change_variable = { name = tot_avmj_interest_arrears_congresses add = 1 } add_modifier = { name = tot_avmj_interest_arrears } set_variable = { name = tot_avmj_creditor_pressure value = 1 } }
	else = { set_variable = { name = tot_avmj_interest_arrears_congresses value = 0 } if = { limit = { has_modifier = tot_avmj_interest_arrears } remove_modifier = tot_avmj_interest_arrears } set_variable = { name = tot_avmj_creditor_pressure value = 0 } }
	set_variable = { name = tot_avmj_debt_fund value = 0 }
}

tot_avmj_finish_assembly = {
	set_variable = { name = tot_avmj_assembly_open value = 0 }
	set_variable = { name = tot_avmj_debt_months_active value = 0 }
	set_variable = { name = tot_avmj_months_until_assembly value = 48 }
	tot_avmj_update_creditor_holders = yes
	tot_avmj_update_debt_stage = yes
}

tot_avmj_update_debt_stage = {
	if = {
		limit = { var:tot_avmj_debt_principal <= 0 var:tot_avmj_debt_stage > 0 }
		set_variable = { name = tot_avmj_debt_stage value = 0 }
		trigger_event = { id = tot_avmj_debt.4 days = 0 popup = yes }
	}
	else_if = {
		limit = { var:tot_avmj_debt_principal <= 420000 var:tot_avmj_debt_stage > 1 }
		set_variable = { name = tot_avmj_debt_stage value = 1 }
		trigger_event = { id = tot_avmj_debt.3 days = 0 popup = yes }
	}
	else_if = {
		limit = { var:tot_avmj_debt_principal <= 840000 var:tot_avmj_debt_stage > 2 }
		set_variable = { name = tot_avmj_debt_stage value = 2 }
		trigger_event = { id = tot_avmj_debt.2 days = 0 popup = yes }
	}
	else_if = {
		limit = { var:tot_avmj_debt_principal <= 1260000 var:tot_avmj_debt_stage > 3 }
		set_variable = { name = tot_avmj_debt_stage value = 3 }
		trigger_event = { id = tot_avmj_debt.1 days = 0 popup = yes }
	}
}

tot_avmj_debt_monthly_process = {
	tot_avmj_setup_debt = yes
	tot_avmj_update_creditor_holders = yes
	tot_avmj_apply_creditor_privileges = yes
	if = {
		limit = { var:tot_avmj_debt_principal > 0 }
		tot_avmj_collect_monthly_repayment = yes
		change_variable = { name = tot_avmj_interest_due add = var:tot_avmj_monthly_interest_payment }
		change_variable = { name = tot_avmj_debt_months_active add = 1 }
		change_variable = { name = tot_avmj_months_until_assembly add = -1 }
		if = {
			limit = { var:tot_avmj_debt_months_active >= 48 var:tot_avmj_assembly_open = 0 }
			set_variable = { name = tot_avmj_assembly_open value = 1 }
			trigger_event = { id = tot_avmj_debt.8 days = 0 popup = yes }
		}
	}
	tot_avmj_refresh_creditor_gui_lists = yes
}
