tot_character_growth_weekly_update = {
	every_scope_character = {
		limit = {
			is_character_alive = yes
			historical = yes
		}

		change_variable = {
			name = tot_character_growth_exp
			add = 10
		}
		if = {
			limit = {
				is_ruler_of_own_country = yes
			}
			change_variable = {
				name = tot_character_growth_exp
				add = 10
			}
		}
		if = {
			limit = {
				interest_group ?= {
					is_interest_group_type = ig_landowners
				}
			}
			change_variable = {
				name = tot_character_growth_exp
				add = 5
			}
		}
		tot_character_growth_check_level_up = yes
	}
}
