tot_character_growth_weekly_update = {
	set_variable = { name = tot_club_growth_bonus value = 0 }
	if = {
		limit = { modifier:country_tot_club_growth_exp_add > 0 }
		set_variable = {
			name = tot_club_growth_bonus
			value = modifier:country_tot_club_growth_exp_add
		}
	}

	every_scope_character = {
		limit = {
			is_character_alive = yes
			historical = yes
		}
		if = {
			limit = { has_variable = tot_character_growth_club_cost_mult }
			remove_variable = tot_character_growth_club_cost_mult
		}
		if = {
			limit = { modifier:country_tot_character_growth_cost_mult_add != 0 }
			set_variable = {
				name = tot_character_growth_club_cost_mult
				value = {
					value = modifier:country_tot_character_growth_cost_mult_add
					multiply = 100
				}
			}
		}

		change_variable = {
			name = tot_character_growth_exp
			add = 10
		}
		if = {
			limit = {
				is_ruler_of_own_country = yes
			}
			change_variable = {
				name = tot_character_growth_exp
				add = 3
			}
		}
		if = {
			limit = {
				is_interest_group_leader = yes
			}
			change_variable = {
				name = tot_character_growth_exp
				add = 3
			}
		}
		if = {
			limit = { root.var:tot_club_growth_bonus > 0 }
			change_variable = {
				name = tot_character_growth_exp
				add = root.var:tot_club_growth_bonus
			}
		}
		tot_character_growth_refresh_cost = yes
		tot_character_growth_check_level_up = yes
	}
}

tot_character_growth_refresh_cost = {
	set_variable = {
		name = tot_character_growth_current_cost
		value = 1000
	}

	if = {
		limit = { has_variable = tot_character_growth_trait_cost_mult }
		change_variable = {
			name = tot_character_growth_current_cost
			multiply = 100
		}
		change_variable = {
			name = tot_character_growth_current_cost
			add = var:tot_character_growth_trait_cost_mult
		}
		change_variable = {
			name = tot_character_growth_current_cost
			divide = 100
		}
	}
	if = {
		limit = { has_variable = tot_character_growth_club_cost_mult }
		change_variable = {
			name = tot_character_growth_current_cost
			multiply = 100
		}
		change_variable = {
			name = tot_character_growth_current_cost
			add = var:tot_character_growth_club_cost_mult
		}
		change_variable = {
			name = tot_character_growth_current_cost
			divide = 100
		}
	}

	if = {
		limit = { has_variable = tot_character_growth_trait_cost_add }
		change_variable = {
			name = tot_character_growth_current_cost
			add = var:tot_character_growth_trait_cost_add
		}
	}

	if = {
		limit = { tot_character_growth_acquired_traits_count_value > 0 }
		change_variable = {
			name = tot_character_growth_current_cost
			add = {
				value = tot_character_growth_acquired_traits_count_value
				multiply = 3000
			}
		}
	}

	if = {
		limit = { var:tot_character_growth_current_cost < 1 }
		set_variable = { name = tot_character_growth_current_cost value = 1 }
	}
}

tot_character_growth_check_level_up = {
	tot_character_growth_refresh_cost = yes

	if = {
		limit = {
			has_variable = tot_character_growth_exp
			has_variable = tot_character_growth_current_cost
			var:tot_character_growth_exp >= var:tot_character_growth_current_cost
			historical = yes
			tot_character_growth_has_available_random_trait = yes
		}

		change_variable = {
			name = tot_character_growth_exp
			add = {
				value = var:tot_character_growth_current_cost
				multiply = -1
			}
		}
		tot_character_growth_grant_random_kizuna_trait = yes
		tot_character_growth_refresh_cost = yes
		owner ?= {
			update_kizuna = yes
		}
	}
}
