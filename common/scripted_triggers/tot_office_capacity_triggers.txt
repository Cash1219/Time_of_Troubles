tot_core_office_capacity_available = {
	OR = {
		AND = { is_historical = yes owner ?= { tot_club_capacity_free >= tot_core_office_maintenance_historical } }
		AND = { is_historical = no owner ?= { tot_club_capacity_free >= tot_core_office_maintenance_standard } }
		# Replacing a holder is allowed only when their released maintenance covers the new holder.
		AND = {
			is_historical = yes
			owner ?= { has_variable = tot_$SLOT$ var:tot_$SLOT$ ?= { is_historical = yes } }
		}
		AND = {
			is_historical = yes
			owner ?= { has_variable = tot_$SLOT$ var:tot_$SLOT$ ?= { is_historical = no } }
		}
		AND = {
			is_historical = no
			owner ?= { has_variable = tot_$SLOT$ var:tot_$SLOT$ ?= { is_historical = no } }
		}
		AND = {
			is_historical = no
			owner ?= {
				tot_club_capacity_free >= tot_core_office_maintenance_difference
				has_variable = tot_$SLOT$
				var:tot_$SLOT$ ?= { is_historical = yes }
			}
		}
	}
}
