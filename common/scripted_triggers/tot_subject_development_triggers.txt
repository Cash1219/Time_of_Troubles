# 共同目标（双方通用）

tot_subject_development_common_1_trigger = {
	OR = {
		AND = {
			country_has_building_type_levels = { target = bt:building_food_industry value >= 3 }
			country_has_building_type_levels = { target = bt:building_textile_mill value >= 3 }
		}
		AND = {
			country_has_building_type_levels = { target = bt:building_tooling_workshop value >= 3 }
			country_has_building_type_levels = { target = bt:building_paper_mill value >= 3 }
		}
		AND = {
			country_has_building_type_levels = { target = bt:building_chemical_plant value >= 3 }
			country_has_building_type_levels = { target = bt:building_glassworks value >= 3 }
		}
	}
}

}

	OR = {

	OR = {
		average_sol >= 12

		}
	}
}
